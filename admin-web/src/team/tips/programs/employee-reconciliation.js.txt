function detailNumber(value) { return Number(value) || 0; }
    var employeeDetailViewState = null;
    var employeeDetailEmailSnapshot = null;
    var employeeDetailPreparedEmailContent = null;
    function detailMoney(value) {
      var number = detailNumber(value);
      return '$' + Math.abs(number).toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ',');
    }
    function detailHours(value) {
      var number = detailNumber(value);
      return (Math.round(number * 100) / 100).toString();
    }
    function employeeDetailRoleLabel(role) {
      return String(role || '').trim() || '未设置角色';
    }
    function employeeDetailAttendanceStatus(row) {
      return row && row.clockStatus === '已打卡' ? '已打卡' : '未打卡';
    }
    function normalizeEmployeeDetailRange(start, end, changedSide) {
      if (start <= end) return { start: start, end: end };
      return changedSide === 'end'
        ? { start: end, end: end }
        : { start: start, end: start };
    }
    function filterEmployeeDetailRows(rows, start, end, attendance) {
      return (rows || []).filter(function(row) {
        var inDateRange = row.dateKey >= start && row.dateKey <= end;
        return inDateRange && (!attendance || attendance === '全部状态' || employeeDetailAttendanceStatus(row) === attendance);
      });
    }
    function summarizeEmployeeDetailRows(rows) {
      return (rows || []).reduce(function(total, row) {
        total.shifts += detailNumber(row.hours) > 0 ? 1 : 0;
        total.hours += detailNumber(row.hours);
        total.before += detailNumber(row.before);
        total.deducted += detailNumber(row.deducted);
        total.received += detailNumber(row.received);
        total.after += detailNumber(row.after);
        return total;
      }, { shifts: 0, hours: 0, before: 0, deducted: 0, received: 0, after: 0 });
    }
    function detailDateLabel(dateKey) {
      var parts = String(dateKey || '').split('-');
      return parts.length === 3 ? parts[1] + '/' + parts[2] : dateKey;
    }
    function detailWeekday(dateKey) {
      var date = new Date(dateKey + 'T00:00:00');
      return isNaN(date.getTime()) ? '' : date.toLocaleDateString('zh-CN', { weekday: 'short' });
    }
    function setDetailText(id, value) {
      var element = document.getElementById(id);
      if (element) element.textContent = value;
    }
    function makeDetailCell(primary, secondary, className) {
      var cell = document.createElement('td');
      var wrap = document.createElement('span');
      wrap.className = className || 'tipout-detail-cell-stack';
      var strong = document.createElement('strong');
      strong.textContent = primary;
      wrap.appendChild(strong);
      if (secondary !== undefined && secondary !== '') {
        var small = document.createElement('small');
        small.textContent = secondary;
        wrap.appendChild(small);
      }
      cell.appendChild(wrap);
      return cell;
    }
    function makeStatusCell(label, className) {
      var cell = document.createElement('td');
      var status = document.createElement('span');
      status.className = 'tipout-detail-status ' + className;
      status.textContent = label;
      cell.appendChild(status);
      return cell;
    }
    function makeMoneyCell(value, kind) {
      var cell = document.createElement('td');
      var strong = document.createElement('strong');
      if (kind === 'deduct') {
        strong.className = 'tip-amount--deduct';
        strong.textContent = '-' + detailMoney(value);
      } else if (kind === 'receive') {
        strong.className = 'tip-amount--receive';
        strong.textContent = '+' + detailMoney(value);
      } else {
        strong.textContent = detailMoney(value);
      }
      cell.appendChild(strong);
      return cell;
    }

    function employeeDetailCsvCell(value) {
      var text = value === undefined || value === null ? '' : String(value);
      if (/^[=+\-@]/.test(text)) text = "'" + text;
      if (/[",\r\n]/.test(text)) return '"' + text.replace(/"/g, '""') + '"';
      return text;
    }

    function employeeDetailSafeFilename(value) {
      return String(value || '员工对账').replace(/[\\/:*?"<>|]/g, '_').replace(/\s+/g, '_');
    }

    function employeeDetailEscapeHtml(value) {
      return String(value === undefined || value === null ? '' : value)
        .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
    }

    function collectEmployeeDetailExportData() {
      var state = getEmployeeDetailViewState();
      if (!state || !state.rows || state.rows.length === 0) {
        showNotification('当前筛选条件下没有可导出的数据', 'warning');
        return null;
      }
      return {
        name: state.snapshot.name || '员工',
        role: employeeDetailRoleLabel(state.snapshot.role),
        store: state.snapshot.store || '',
        start: state.range.start,
        end: state.range.end,
        attendance: state.attendance,
        rows: state.rows.map(function(row) {
          return {
            date: row.dateKey,
            attendance: employeeDetailAttendanceStatus(row),
            shift: detailNumber(row.hours) > 0 ? '1 个班次 / ' + detailHours(row.hours) + ' h' : '— / 0 h',
            before: detailMoney(row.before),
            deducted: '-' + detailMoney(row.deducted),
            received: '+' + detailMoney(row.received),
            after: detailMoney(row.after),
            allocation: row.allocated ? '已分配' : '待分配'
          };
        })
      };
    }

    function employeeDetailExportFilename(data, extension) {
      return employeeDetailSafeFilename(data.name + '_' + data.start + '_' + data.end) + '.' + extension;
    }

    function buildEmployeeDetailCsv(data) {
      var lines = [];
      lines.push(['员工', data.name].map(employeeDetailCsvCell).join(','));
      lines.push(['角色', data.role].map(employeeDetailCsvCell).join(','));
      lines.push(['日期范围', data.start + ' ~ ' + data.end].map(employeeDetailCsvCell).join(','));
      lines.push(['考勤筛选', data.attendance].map(employeeDetailCsvCell).join(','));
      lines.push('');
      lines.push(['日期', '考勤状态', '班次 / 工时', '分配前', '扣除', '分配获得', '实际获得', '分配状态'].map(employeeDetailCsvCell).join(','));
      data.rows.forEach(function(row) {
        lines.push([row.date, row.attendance, row.shift, row.before, row.deducted, row.received, row.after, row.allocation].map(employeeDetailCsvCell).join(','));
      });
      return '\uFEFF' + lines.join('\r\n');
    }

    function exportEmployeeDetailCsv(data) {
      var blob = new Blob([buildEmployeeDetailCsv(data)], { type: 'text/csv;charset=utf-8;' });
      var url = URL.createObjectURL(blob);
      var link = document.createElement('a');
      link.href = url;
      link.download = employeeDetailExportFilename(data, 'csv');
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      URL.revokeObjectURL(url);
      showNotification('CSV 导出成功', 'success');
    }

    function buildEmployeeDetailPrintHtml(data) {
      var headings = ['日期', '考勤状态', '班次 / 工时', '分配前', '扣除', '分配获得', '实际获得', '分配状态'];
      var body = data.rows.map(function(row) {
        return '<tr>' + [row.date, row.attendance, row.shift, row.before, row.deducted, row.received, row.after, row.allocation]
          .map(function(value) { return '<td>' + employeeDetailEscapeHtml(value) + '</td>'; }).join('') + '</tr>';
      }).join('');
      return '<!doctype html><html lang="zh-CN"><head><meta charset="utf-8"><title>' +
        employeeDetailEscapeHtml(employeeDetailExportFilename(data, 'pdf').replace(/\.pdf$/, '')) +
        '</title><style>@page{size:A4 landscape;margin:12mm}body{font-family:"Microsoft YaHei","PingFang SC",Arial,sans-serif;color:#171717;font-size:12px}h1{font-size:20px;margin:0 0 8px}.meta{color:#666;margin-bottom:16px}table{width:100%;border-collapse:collapse}th,td{border:1px solid #ddd;padding:7px;text-align:left}th{background:#f5f6f8}</style></head><body><h1>' +
        employeeDetailEscapeHtml(data.name) + ' 员工对账</h1><div class="meta">角色：' + employeeDetailEscapeHtml(data.role) +
        '　日期：' + employeeDetailEscapeHtml(data.start + ' ～ ' + data.end) + '　考勤：' + employeeDetailEscapeHtml(data.attendance) +
        '</div><table><thead><tr>' + headings.map(function(item) { return '<th>' + item + '</th>'; }).join('') +
        '</tr></thead><tbody>' + body + '</tbody></table></body></html>';
    }

    function exportEmployeeDetailPdf(data) {
      var printWindow = window.open('', '_blank', 'noopener,noreferrer');
      if (!printWindow) {
        showNotification('浏览器阻止了打印窗口，请允许弹窗后重试', 'error');
        return;
      }
      printWindow.document.open();
      printWindow.document.write(buildEmployeeDetailPrintHtml(data));
      printWindow.document.close();
      printWindow.focus();
      printWindow.print();
    }

    function closeEmployeeDetailExportMenu() {
      var menu = document.getElementById('employeeDetailExportMenu');
      var button = document.getElementById('employeeDetailExportButton');
      if (menu) menu.classList.remove('show');
      if (button) button.setAttribute('aria-expanded', 'false');
    }

    function toggleEmployeeDetailExportMenu(event) {
      if (event && event.stopPropagation) event.stopPropagation();
      var menu = document.getElementById('employeeDetailExportMenu');
      var button = document.getElementById('employeeDetailExportButton');
      if (!menu) return;
      var open = !menu.classList.contains('show');
      menu.classList.toggle('show', open);
      if (button) button.setAttribute('aria-expanded', open ? 'true' : 'false');
    }

    function runEmployeeDetailExport(type) {
      closeEmployeeDetailExportMenu();
      var data = collectEmployeeDetailExportData();
      if (!data) return;
      if (type === 'pdf') exportEmployeeDetailPdf(data);
      if (type === 'csv') exportEmployeeDetailCsv(data);
    }

    function parseEmployeeDetailEmails(value) {
      var emails = String(value || '').split(',').map(function(email) { return email.trim(); }).filter(Boolean);
      var emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (emails.length === 0 || emails.some(function(email) { return !emailPattern.test(email); })) return null;
      return emails;
    }

    function openEmployeeDetailEmailModal() {
      closeEmployeeDetailExportMenu();
      var data = collectEmployeeDetailExportData();
      if (!data) return;
      employeeDetailEmailSnapshot = JSON.parse(JSON.stringify(data));
      employeeDetailPreparedEmailContent = null;
      openModal('employeeDetailEmailModal');
    }

    function sendEmployeeDetailEmail() {
      var emailInput = document.getElementById('employeeDetailExportEmail');
      var emails = parseEmployeeDetailEmails(emailInput ? emailInput.value : '');
      if (!emails) {
        showNotification('请输入正确的邮箱地址，多个邮箱请用逗号分隔', 'error');
        return;
      }
      if (!employeeDetailEmailSnapshot) {
        showNotification('导出内容已失效，请重新打开邮箱发送', 'error');
        closeModal('employeeDetailEmailModal');
        return;
      }
      var formatInput = document.querySelector('input[name="employeeDetailEmailFormat"]:checked');
      var format = formatInput ? formatInput.value : '';
      if (format !== 'pdf' && format !== 'csv') {
        showNotification('请选择导出格式', 'error');
        return;
      }
      employeeDetailPreparedEmailContent = format === 'pdf'
        ? buildEmployeeDetailPrintHtml(employeeDetailEmailSnapshot)
        : buildEmployeeDetailCsv(employeeDetailEmailSnapshot);
      closeModal('employeeDetailEmailModal');
      showNotification(format.toUpperCase() + ' 已提交发送至 ' + emails.join(', ') + '（当前为模拟发送）', 'success');
    }

    function renderEmployeeDetailRows(rows) {
      var tbody = document.getElementById('employeeDetailRows');
      tbody.textContent = '';
      rows.forEach(function(row) {
        var tr = document.createElement('tr');
        var attendanceStatus = employeeDetailAttendanceStatus(row);
        var attendanceMissing = attendanceStatus === '未打卡';
        var hasIssue = attendanceMissing || !row.allocated;
        if (hasIssue) tr.className = 'has-issue';
        tr.appendChild(makeDetailCell(detailDateLabel(row.dateKey), detailWeekday(row.dateKey), 'tipout-daily-date'));
        tr.appendChild(makeStatusCell('● ' + attendanceStatus, attendanceMissing ? 'is-pending' : 'is-complete'));
        tr.appendChild(makeDetailCell(detailNumber(row.hours) > 0 ? '1 个班次' : '—', detailHours(row.hours) + ' h', 'tipout-shift-summary'));
        tr.appendChild(makeMoneyCell(row.before));
        tr.appendChild(makeMoneyCell(row.deducted, 'deduct'));
        tr.appendChild(makeMoneyCell(row.received, 'receive'));
        tr.appendChild(makeMoneyCell(row.after));
        tr.appendChild(makeStatusCell(row.allocated ? '已分配' : '待分配', row.allocated ? 'is-allocated' : 'is-pending'));
        tbody.appendChild(tr);
      });
    }

    function renderEmployeeDetailSummary(summary) {
      summary = summary || {};
      setDetailText('employeeDetailShifts', String(detailNumber(summary.shifts)));
      setDetailText('employeeDetailHours', detailHours(summary.hours) + ' h');
      setDetailText('employeeDetailBefore', detailMoney(summary.before));
      setDetailText('employeeDetailDeducted', '-' + detailMoney(summary.deducted));
      setDetailText('employeeDetailReceived', '+' + detailMoney(summary.received));
      setDetailText('employeeDetailAfter', detailMoney(summary.after));
    }

    function renderEmployeeDetailRange(snapshot, changedSide) {
      var startInput = document.getElementById('employeeDetailStartDate');
      var endInput = document.getElementById('employeeDetailEndDate');
      var attendanceInput = document.getElementById('employeeDetailAttendanceFilter');
      var range = normalizeEmployeeDetailRange(startInput.value, endInput.value, changedSide);
      startInput.value = range.start;
      endInput.value = range.end;
      var attendance = attendanceInput ? attendanceInput.value : '全部状态';
      var rows = filterEmployeeDetailRows(snapshot.dailyRows, range.start, range.end, attendance);
      employeeDetailViewState = { snapshot: snapshot, range: range, attendance: attendance, rows: rows.slice() };
      renderEmployeeDetailSummary(summarizeEmployeeDetailRows(rows));
      renderEmployeeDetailRows(rows);
      var table = document.getElementById('employeeDetailTable');
      var filteredEmpty = document.getElementById('employeeDetailFilteredEmpty');
      if (table) table.hidden = rows.length === 0;
      if (filteredEmpty) filteredEmpty.hidden = rows.length !== 0;
    }

    function getEmployeeDetailViewState() {
      return employeeDetailViewState;
    }

    function renderEmployeeReconciliationDetail(snapshot) {
      setDetailText('employeeDetailName', snapshot.name || '—');
      setDetailText('employeeDetailRole', employeeDetailRoleLabel(snapshot.role));
      var startInput = document.getElementById('employeeDetailStartDate');
      var endInput = document.getElementById('employeeDetailEndDate');
      var attendanceInput = document.getElementById('employeeDetailAttendanceFilter');
      startInput.min = snapshot.dateStart;
      startInput.max = snapshot.dateEnd;
      endInput.min = snapshot.dateStart;
      endInput.max = snapshot.dateEnd;
      startInput.value = snapshot.dateStart;
      endInput.value = snapshot.dateEnd;
      startInput.addEventListener('change', function() { renderEmployeeDetailRange(snapshot, 'start'); });
      endInput.addEventListener('change', function() { renderEmployeeDetailRange(snapshot, 'end'); });
      if (attendanceInput) attendanceInput.addEventListener('change', function() { renderEmployeeDetailRange(snapshot, 'attendance'); });
      renderEmployeeDetailRange(snapshot, 'start');
    }

    function showEmployeeDetailEmpty() {
      document.getElementById('employeeDetailContent').hidden = true;
      document.getElementById('employeeDetailEmpty').hidden = false;
    }

    function returnToEmployeeReconciliation() {
      var params = new URLSearchParams(window.location.search);
      if (params.get('from') === 'summary' && params.get('return') === 'history' && history.length > 1) {
        history.back();
        return;
      }
      window.location.href = 'index.html?view=employee';
    }

    function initializeEmployeeReconciliationDetail() {
      var params = new URLSearchParams(window.location.search);
      var detailContext = {
        employeeId: params.get('employeeId') || '',
        store: params.get('store') || '',
        dateStart: params.get('start') || '',
        dateEnd: params.get('end') || ''
      };
      var snapshot = TipOutSummaryUi.readEmployeeReconciliationSnapshot(
        sessionStorage.getItem(TipOutSummaryUi.EMPLOYEE_RECONCILIATION_SNAPSHOT_KEY),
        detailContext
      );
      if (!snapshot) {
        showEmployeeDetailEmpty();
        return;
      }
      renderEmployeeReconciliationDetail(snapshot);
    }

    document.addEventListener('click', function(event) {
      if (!event.target || !event.target.closest || !event.target.closest('.export-dropdown')) closeEmployeeDetailExportMenu();
    });
    document.addEventListener('DOMContentLoaded', initializeEmployeeReconciliationDetail);
