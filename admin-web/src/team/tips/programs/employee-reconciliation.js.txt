function detailNumber(value) { return Number(value) || 0; }
    var employeeDetailViewState = null;
    var employeeDetailDateSort = 'desc';
    var employeeDetailEmailSnapshot = null;
    var employeeDetailPreparedEmailContent = null;
    function detailMoney(value) {
      var number = detailNumber(value);
      return '$' + Math.abs(number).toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ',');
    }
    function detailHours(value) {
      var number = detailNumber(value);
      return (Math.round(number * 100) / 100).toString();
    }
    function employeeDetailRoleLabel(role) {
      return String(role || '').trim() || '未设置角色';
    }
    function employeeDetailAttendanceSummary(row) {
      if (window.TipOutAttendance && window.TipOutAttendance.summarizeDayAttendance) {
        return window.TipOutAttendance.summarizeDayAttendance(row);
      }
      var hours = detailNumber(row && row.hours);
      if (row && row.clockStatus === '已打卡' && hours > 0) return { status: '已打卡', shifts: 1, hours: hours, clockIns: ['—'], clockOuts: ['—'] };
      if (row && row.clockStatus === '已打卡') return { status: '打卡异常', shifts: 0, hours: 0, clockIns: ['—'], clockOuts: ['—'] };
      return { status: '未打卡', shifts: 0, hours: 0, clockIns: ['—'], clockOuts: ['—'] };
    }
    function employeeDetailAttendanceStatus(row) {
      return employeeDetailAttendanceSummary(row).status;
    }
    function normalizeEmployeeDetailRange(start, end, changedSide) {
      if (start <= end) return { start: start, end: end };
      return changedSide === 'end'
        ? { start: end, end: end }
        : { start: start, end: start };
    }
    function filterEmployeeDetailRows(rows, start, end, attendance, allocationStatus) {
      return (rows || []).filter(function(row) {
        var inDateRange = row.dateKey >= start && row.dateKey <= end;
        var attendanceMatches = !attendance || attendance === '全部状态' || employeeDetailAttendanceStatus(row) === attendance;
        var rowAllocationStatus = row.allocated ? '已分配' : '待分配';
        var allocationMatches = !allocationStatus || allocationStatus === '全部状态' || rowAllocationStatus === allocationStatus;
        return inDateRange && attendanceMatches && allocationMatches;
      });
    }
    function summarizeEmployeeDetailRows(rows) {
      var summary = (rows || []).reduce(function(total, row) {
        var attendance = employeeDetailAttendanceSummary(row);
        var punchValid = row.punchHoursValid === undefined ? attendance.status !== '未打卡' : !!row.punchHoursValid;
        total.recordDays += 1;
        if (punchValid) { total.punchValidDays += 1; total.punchHours += detailNumber(row.punchHours === undefined ? attendance.hours : row.punchHours); }
        (attendance.manualEntries || []).forEach(function(entry) { total.manualPools[entry.poolId] = true; total.manualRecords += 1; });
        total.before += detailNumber(row.before);
        if (row.allocated) {
          total.allocatedCount += 1;
          total.deducted += detailNumber(row.deducted);
          total.received += detailNumber(row.received);
          total.after += detailNumber(row.after);
        }
        return total;
      }, { punchHours: 0, punchValidDays: 0, recordDays: 0, manualPools: {}, manualRecords: 0, before: 0, deducted: 0, received: 0, after: 0, allocatedCount: 0 });
      summary.manualPoolCount = Object.keys(summary.manualPools).length;
      summary.punchHoursDisplay = summary.punchValidDays
        ? detailHours(summary.punchHours) + ' h' + (summary.punchValidDays < summary.recordDays ? '（' + summary.punchValidDays + '/' + summary.recordDays + ' 天有记录）' : '')
        : '—';
      summary.allocationHourSummaries = typeof TipOutSummaryUi !== 'undefined' && TipOutSummaryUi.aggregateAllocationHourEntries
        ? TipOutSummaryUi.aggregateAllocationHourEntries(rows) : [];
      return summary;
    }
    function detailDateLabel(dateKey) {
      var parts = String(dateKey || '').split('-');
      return parts.length === 3 ? parts[1] + '/' + parts[2] : dateKey;
    }
    function detailWeekday(dateKey) {
      var date = new Date(dateKey + 'T00:00:00');
      return isNaN(date.getTime()) ? '' : date.toLocaleDateString('zh-CN', { weekday: 'short' });
    }
    function setDetailText(id, value) {
      var element = document.getElementById(id);
      if (element) element.textContent = value;
    }
    function makeDetailCell(primary, secondary, className) {
      var cell = document.createElement('td');
      var wrap = document.createElement('span');
      wrap.className = className || 'tipout-detail-cell-stack';
      var strong = document.createElement('strong');
      strong.textContent = primary;
      wrap.appendChild(strong);
      if (secondary !== undefined && secondary !== '') {
        var small = document.createElement('small');
        small.textContent = secondary;
        wrap.appendChild(small);
      }
      cell.appendChild(wrap);
      return cell;
    }
    function makeStatusCell(label, className) {
      var cell = document.createElement('td');
      var status = document.createElement('span');
      status.className = 'tipout-detail-status ' + className;
      status.textContent = label;
      cell.appendChild(status);
      return cell;
    }
    function makeWorkHoursCell(lines) {
      var cell = document.createElement('td');
      var stack = document.createElement('span');
      stack.className = 'tipout-work-hours-stack';
      (lines || []).forEach(function(line) {
        var row = document.createElement('span');
        row.className = 'tipout-work-hours-line';
        var value = document.createElement('strong');
        value.textContent = detailHours(line.hours) + ' h';
        row.appendChild(value);
        if (line.source === 'manual') {
          var badge = document.createElement('span');
          badge.className = 'tipout-hours-entry-badge';
          badge.textContent = '录入';
          row.appendChild(badge);
        }
        stack.appendChild(row);
      });
      cell.appendChild(stack);
      return cell;
    }
    function makePunchHoursCell(row, attendance) {
      var valid = row.punchHoursValid === undefined ? attendance.status !== '未打卡' : !!row.punchHoursValid;
      return makeDetailCell(valid ? detailHours(row.punchHours === undefined ? attendance.hours : row.punchHours) + ' h' : '—');
    }
    function makeAllocationHoursCell(row) {
      var cell = document.createElement('td');
      var stack = document.createElement('span');
      stack.className = 'tipout-allocation-hours-stack';
      var punch = row.punchHoursValid === false ? null : row.punchHours;
      var entries = TipOutSummaryUi.normalizeEmployeeHoursRow(row).allocationHourEntries;
      if (!entries.length) entries = [{ label: '', usesHours: true, hoursValid: false }];
      entries.forEach(function(entry) {
        var line = document.createElement('span');
        line.className = 'tipout-allocation-hours-line' + (TipOutSummaryUi.shouldHighlightAllocationHours(punch, entry) ? ' tipout-allocation-hours--changed' : '');
        var label = entry.poolName && entry.ruleName ? entry.poolName + ' · ' + entry.ruleName + ' ' : '';
        line.textContent = label + (entry.usesHours && entry.hoursValid ? detailHours(entry.hours) + ' h' : '—');
        stack.appendChild(line);
      });
      cell.appendChild(stack);
      return cell;
    }
    function makeMoneyCell(value, kind) {
      var cell = document.createElement('td');
      var strong = document.createElement('strong');
      if (kind === 'deduct') {
        strong.className = 'tip-amount--deduct';
        strong.textContent = '-' + detailMoney(value);
      } else if (kind === 'receive') {
        strong.className = 'tip-amount--receive';
        strong.textContent = '+' + detailMoney(value);
      } else {
        strong.textContent = detailMoney(value);
      }
      cell.appendChild(strong);
      return cell;
    }
    function makeAllocationMoneyCell(row, value, kind) {
      return row && row.allocated ? makeMoneyCell(value, kind) : makeDetailCell('—');
    }
    function employeeDetailAllocationMoney(row, value, kind) {
      if (!row || !row.allocated) return '—';
      if (kind === 'deduct') return '-' + detailMoney(value);
      if (kind === 'receive') return '+' + detailMoney(value);
      return detailMoney(value);
    }

    function employeeDetailCsvCell(value) {
      var text = value === undefined || value === null ? '' : String(value);
      if (/^[=+\-@]/.test(text)) text = "'" + text;
      if (/[",\r\n]/.test(text)) return '"' + text.replace(/"/g, '""') + '"';
      return text;
    }

    function employeeDetailSafeFilename(value) {
      return String(value || '员工分配汇总').replace(/[\\/:*?"<>|]/g, '_').replace(/\s+/g, '_');
    }

    function employeeDetailEscapeHtml(value) {
      return String(value === undefined || value === null ? '' : value)
        .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
    }

    function collectEmployeeDetailExportData() {
      var state = getEmployeeDetailViewState();
      if (!state || !state.rows || state.rows.length === 0) {
        showNotification('当前筛选条件下没有可导出的数据', 'warning');
        return null;
      }
      return {
        name: state.snapshot.name || '员工',
        role: employeeDetailRoleLabel(state.snapshot.role),
        store: state.snapshot.store || '',
        start: state.range.start,
        end: state.range.end,
        attendance: state.attendance,
        allocationStatus: state.allocationStatus,
        rows: state.rows.map(function(row) {
          var attendance = employeeDetailAttendanceSummary(row);
          return {
            date: row.dateKey,
            attendance: attendance.status,
            workHourSource: (attendance.manualEntries || []).length ? (attendance.punchStatus === '未打卡' ? '手动工时' : '混合工时') : '打卡工时',
            attendanceNote: attendance.hasPunchException ? '打卡异常' : '',
            punchHours: row.punchHoursValid === false || (row.punchHoursValid === undefined && attendance.status === '未打卡') ? '—' : detailHours(row.punchHours === undefined ? attendance.hours : row.punchHours) + ' h',
            allocationHours: TipOutSummaryUi.normalizeEmployeeHoursRow(row).allocationHourEntries.map(function(entry) {
              return entry.poolName + ' · ' + entry.ruleName + ': ' + (entry.usesHours && entry.hoursValid ? detailHours(entry.hours) + ' h' : '—');
            }).join('\n') || '—',
            manualHourDetails: (attendance.manualEntries || []).map(function(entry) { return entry.poolName + ' · ' + entry.ruleName + ': ' + detailHours(entry.hours) + ' h'; }).join('\n'),
            before: detailMoney(row.before),
            deducted: employeeDetailAllocationMoney(row, row.deducted, 'deduct'),
            received: employeeDetailAllocationMoney(row, row.received, 'receive'),
            after: employeeDetailAllocationMoney(row, row.after),
            allocation: row.allocated ? '已分配' : '待分配'
          };
        })
      };
    }

    function employeeDetailExportFilename(data, extension) {
      return employeeDetailSafeFilename(data.name + '_' + data.start + '_' + data.end) + '.' + extension;
    }

    function buildEmployeeDetailCsv(data) {
      var lines = [];
      lines.push(['员工', data.name].map(employeeDetailCsvCell).join(','));
      lines.push(['角色', data.role].map(employeeDetailCsvCell).join(','));
      lines.push(['日期范围', data.start + ' ~ ' + data.end].map(employeeDetailCsvCell).join(','));
      lines.push(['考勤筛选', data.attendance].map(employeeDetailCsvCell).join(','));
      lines.push(['分配状态筛选', data.allocationStatus].map(employeeDetailCsvCell).join(','));
      lines.push('');
      lines.push(['日期', '考勤状态', '工时来源', '考勤备注', '打卡工时', '分配工时', '手动工时明细', '分配前', '扣除', '分配获得', '实际获得', '分配状态'].map(employeeDetailCsvCell).join(','));
      data.rows.forEach(function(row) {
        lines.push([row.date, row.attendance, row.workHourSource, row.attendanceNote, row.punchHours, row.allocationHours, row.manualHourDetails, row.before, row.deducted, row.received, row.after, row.allocation].map(employeeDetailCsvCell).join(','));
      });
      return '\uFEFF' + lines.join('\r\n');
    }

    function exportEmployeeDetailCsv(data) {
      var blob = new Blob([buildEmployeeDetailCsv(data)], { type: 'text/csv;charset=utf-8;' });
      var url = URL.createObjectURL(blob);
      var link = document.createElement('a');
      link.href = url;
      link.download = employeeDetailExportFilename(data, 'csv');
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      URL.revokeObjectURL(url);
      showNotification('CSV 导出成功', 'success');
    }

    function buildEmployeeDetailPrintHtml(data) {
      var headings = ['日期', '考勤状态', '工时来源', '考勤备注', '打卡工时', '分配工时', '手动工时明细', '分配前', '扣除', '分配获得', '实际获得', '分配状态'];
      var body = data.rows.map(function(row) {
        return '<tr>' + [row.date, row.attendance, row.workHourSource, row.attendanceNote, row.punchHours, row.allocationHours, row.manualHourDetails, row.before, row.deducted, row.received, row.after, row.allocation]
          .map(function(value) { return '<td>' + employeeDetailEscapeHtml(value) + '</td>'; }).join('') + '</tr>';
      }).join('');
      return '<!doctype html><html lang="zh-CN"><head><meta charset="utf-8"><title>' +
        employeeDetailEscapeHtml(employeeDetailExportFilename(data, 'pdf').replace(/\.pdf$/, '')) +
        '</title><style>@page{size:A4 landscape;margin:12mm}body{font-family:"Microsoft YaHei","PingFang SC",Arial,sans-serif;color:#171717;font-size:12px}h1{font-size:20px;margin:0 0 8px}.meta{color:#666;margin-bottom:16px}table{width:100%;border-collapse:collapse}th,td{border:1px solid #ddd;padding:7px;text-align:left;white-space:pre-line}th{background:#f5f6f8}</style></head><body><h1>' +
        employeeDetailEscapeHtml(data.name) + ' 员工分配汇总</h1><div class="meta">角色：' + employeeDetailEscapeHtml(data.role) +
        '　日期：' + employeeDetailEscapeHtml(data.start + ' ～ ' + data.end) + '　考勤：' + employeeDetailEscapeHtml(data.attendance) +
        '　分配状态：' + employeeDetailEscapeHtml(data.allocationStatus) +
        '</div><table><thead><tr>' + headings.map(function(item) { return '<th>' + item + '</th>'; }).join('') +
        '</tr></thead><tbody>' + body + '</tbody></table></body></html>';
    }

    function exportEmployeeDetailPdf(data) {
      var printWindow = window.open('', '_blank');
      if (!printWindow) {
        showNotification('浏览器阻止了打印窗口，请允许弹窗后重试', 'error');
        return;
      }
      printWindow.opener = null;
      printWindow.document.open();
      printWindow.document.write(buildEmployeeDetailPrintHtml(data));
      printWindow.document.close();
      printWindow.focus();
      printWindow.print();
    }

    function closeEmployeeDetailExportMenu() {
      var menu = document.getElementById('employeeDetailExportMenu');
      var button = document.getElementById('employeeDetailExportButton');
      if (menu) menu.classList.remove('show');
      if (button) button.setAttribute('aria-expanded', 'false');
    }

    function toggleEmployeeDetailExportMenu(event) {
      if (event && event.stopPropagation) event.stopPropagation();
      var menu = document.getElementById('employeeDetailExportMenu');
      var button = document.getElementById('employeeDetailExportButton');
      if (!menu) return;
      var open = !menu.classList.contains('show');
      menu.classList.toggle('show', open);
      if (button) button.setAttribute('aria-expanded', open ? 'true' : 'false');
    }

    function runEmployeeDetailExport(type) {
      closeEmployeeDetailExportMenu();
      var data = collectEmployeeDetailExportData();
      if (!data) return;
      if (type === 'pdf') exportEmployeeDetailPdf(data);
      if (type === 'csv') exportEmployeeDetailCsv(data);
    }

    function parseEmployeeDetailEmails(value) {
      var emails = String(value || '').split(',').map(function(email) { return email.trim(); }).filter(Boolean);
      var emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (emails.length === 0 || emails.some(function(email) { return !emailPattern.test(email); })) return null;
      return emails;
    }

    function openEmployeeDetailEmailModal() {
      closeEmployeeDetailExportMenu();
      var data = collectEmployeeDetailExportData();
      if (!data) return;
      employeeDetailEmailSnapshot = JSON.parse(JSON.stringify(data));
      employeeDetailPreparedEmailContent = null;
      openModal('employeeDetailEmailModal');
    }

    function sendEmployeeDetailEmail() {
      var emailInput = document.getElementById('employeeDetailExportEmail');
      var emails = parseEmployeeDetailEmails(emailInput ? emailInput.value : '');
      if (!emails) {
        showNotification('请输入正确的邮箱地址，多个邮箱请用逗号分隔', 'error');
        return;
      }
      if (!employeeDetailEmailSnapshot) {
        showNotification('导出内容已失效，请重新打开邮箱发送', 'error');
        closeModal('employeeDetailEmailModal');
        return;
      }
      var formatInput = document.querySelector('input[name="employeeDetailEmailFormat"]:checked');
      var format = formatInput ? formatInput.value : '';
      if (format !== 'pdf' && format !== 'csv') {
        showNotification('请选择导出格式', 'error');
        return;
      }
      employeeDetailPreparedEmailContent = format === 'pdf'
        ? buildEmployeeDetailPrintHtml(employeeDetailEmailSnapshot)
        : buildEmployeeDetailCsv(employeeDetailEmailSnapshot);
      closeModal('employeeDetailEmailModal');
      showNotification(format.toUpperCase() + ' 已提交发送至 ' + emails.join(', ') + '（当前为模拟发送）', 'success');
    }

    function renderEmployeeDetailRows(rows) {
      var tbody = document.getElementById('employeeDetailRows');
      tbody.textContent = '';
      rows.forEach(function(row) {
        var tr = document.createElement('tr');
        var attendance = employeeDetailAttendanceSummary(row);
        var attendanceIssue = attendance.status !== '已打卡';
        var hasIssue = attendanceIssue || !row.allocated;
        if (hasIssue) tr.className = 'has-issue';
        tr.appendChild(makeDetailCell(detailDateLabel(row.dateKey), detailWeekday(row.dateKey), 'tipout-daily-date'));
        tr.appendChild(makeStatusCell('● ' + attendance.status, attendanceIssue ? 'is-pending' : 'is-complete'));
        if (attendance.hasPunchException && attendance.status === '混合工时') tr.cells[1].appendChild(makeStatusCell('打卡异常', 'is-pending').firstChild);
        tr.appendChild(makePunchHoursCell(row, attendance));
        tr.appendChild(makeAllocationHoursCell(row));
        tr.appendChild(makeMoneyCell(row.before));
        tr.appendChild(makeAllocationMoneyCell(row, row.deducted, 'deduct'));
        tr.appendChild(makeAllocationMoneyCell(row, row.received, 'receive'));
        tr.appendChild(makeAllocationMoneyCell(row, row.after));
        tr.appendChild(makeStatusCell(row.allocated ? '已分配' : '待分配', row.allocated ? 'is-allocated' : 'is-pending'));
        tbody.appendChild(tr);
      });
    }

    function renderEmployeeDetailSummary(summary) {
      summary = summary || {};
      setDetailText('employeeDetailPunchHours', summary.punchHoursDisplay);
      setDetailText('employeeDetailAllocationHours', summary.allocationHourSummaries.length ? summary.allocationHourSummaries.map(function(item) { return item.display; }).join(' / ') : '—');
      setDetailText('employeeDetailManualHours', detailNumber(summary.manualPoolCount) + ' 个小费池 / ' + detailNumber(summary.manualRecords) + ' 条记录');
      setDetailText('employeeDetailBefore', detailMoney(summary.before));
      setDetailText('employeeDetailDeducted', summary.allocatedCount ? '-' + detailMoney(summary.deducted) : '—');
      setDetailText('employeeDetailReceived', summary.allocatedCount ? '+' + detailMoney(summary.received) : '—');
      setDetailText('employeeDetailAfter', summary.allocatedCount ? detailMoney(summary.after) : '—');
    }

    function syncEmployeeDetailDateSortHeader() {
      var descending = employeeDetailDateSort === 'desc';
      var header = document.getElementById('employeeDetailDateSortHeader');
      var button = document.getElementById('employeeDetailDateSortButton');
      var icon = document.getElementById('employeeDetailDateSortIcon');
      if (header) header.setAttribute('aria-sort', descending ? 'descending' : 'ascending');
      if (icon) icon.textContent = descending ? '↓' : '↑';
      if (button) button.setAttribute('aria-label', descending
        ? '日期，当前倒序，点击切换为顺序'
        : '日期，当前顺序，点击切换为倒序');
    }

    function toggleEmployeeDetailDateSort() {
      employeeDetailDateSort = employeeDetailDateSort === 'desc' ? 'asc' : 'desc';
      if (employeeDetailViewState && employeeDetailViewState.snapshot) {
        renderEmployeeDetailRange(employeeDetailViewState.snapshot, 'sort');
      } else {
        syncEmployeeDetailDateSortHeader();
      }
    }

    function renderEmployeeDetailRange(snapshot, changedSide) {
      var startInput = document.getElementById('employeeDetailStartDate');
      var endInput = document.getElementById('employeeDetailEndDate');
      var attendanceInput = document.getElementById('employeeDetailAttendanceFilter');
      var allocationStatusInput = document.getElementById('employeeDetailAllocationStatusFilter');
      var range = normalizeEmployeeDetailRange(startInput.value, endInput.value, changedSide);
      startInput.value = range.start;
      endInput.value = range.end;
      var attendance = attendanceInput ? attendanceInput.value : '全部状态';
      var allocationStatus = allocationStatusInput ? allocationStatusInput.value : '全部状态';
      var filteredRows = filterEmployeeDetailRows(snapshot.dailyRows, range.start, range.end, attendance, allocationStatus);
      var rows = TipOutSummaryUi.sortEmployeeDetailRows(filteredRows, employeeDetailDateSort);
      employeeDetailViewState = { snapshot: snapshot, range: range, attendance: attendance, allocationStatus: allocationStatus, rows: rows.slice() };
      syncEmployeeDetailDateSortHeader();
      renderEmployeeDetailSummary(summarizeEmployeeDetailRows(rows));
      renderEmployeeDetailRows(rows);
      var table = document.getElementById('employeeDetailTable');
      var filteredEmpty = document.getElementById('employeeDetailFilteredEmpty');
      if (table) table.hidden = rows.length === 0;
      if (filteredEmpty) filteredEmpty.hidden = rows.length !== 0;
    }

    function getEmployeeDetailViewState() {
      return employeeDetailViewState;
    }

    function renderEmployeeReconciliationDetail(snapshot) {
      employeeDetailDateSort = 'desc';
      setDetailText('employeeDetailName', snapshot.name || '—');
      setDetailText('employeeDetailRole', employeeDetailRoleLabel(snapshot.role));
      var startInput = document.getElementById('employeeDetailStartDate');
      var endInput = document.getElementById('employeeDetailEndDate');
      var attendanceInput = document.getElementById('employeeDetailAttendanceFilter');
      var allocationStatusInput = document.getElementById('employeeDetailAllocationStatusFilter');
      startInput.min = snapshot.dateStart;
      startInput.max = snapshot.dateEnd;
      endInput.min = snapshot.dateStart;
      endInput.max = snapshot.dateEnd;
      startInput.value = snapshot.dateStart;
      endInput.value = snapshot.dateEnd;
      if (allocationStatusInput) allocationStatusInput.value = '全部状态';
      startInput.addEventListener('change', function() { renderEmployeeDetailRange(snapshot, 'start'); });
      endInput.addEventListener('change', function() { renderEmployeeDetailRange(snapshot, 'end'); });
      if (attendanceInput) attendanceInput.addEventListener('change', function() { renderEmployeeDetailRange(snapshot, 'attendance'); });
      if (allocationStatusInput) allocationStatusInput.addEventListener('change', function() { renderEmployeeDetailRange(snapshot, 'allocation'); });
      renderEmployeeDetailRange(snapshot, 'start');
    }

    function showEmployeeDetailEmpty() {
      document.getElementById('employeeDetailContent').hidden = true;
      document.getElementById('employeeDetailEmpty').hidden = false;
    }

    function returnToEmployeeReconciliation() {
      window.location.href = 'index.html?view=employee';
    }

    function initializeEmployeeReconciliationDetail() {
      var params = new URLSearchParams(window.location.search);
      var detailContext = {
        employeeId: params.get('employeeId') || '',
        store: params.get('store') || '',
        dateStart: params.get('start') || '',
        dateEnd: params.get('end') || ''
      };
      var snapshot = TipOutSummaryUi.readEmployeeReconciliationSnapshot(
        sessionStorage.getItem(TipOutSummaryUi.EMPLOYEE_RECONCILIATION_SNAPSHOT_KEY),
        detailContext
      );
      if (!snapshot) {
        showEmployeeDetailEmpty();
        return;
      }
      renderEmployeeReconciliationDetail(snapshot);
    }

    document.addEventListener('click', function(event) {
      if (!event.target || !event.target.closest || !event.target.closest('.export-dropdown')) closeEmployeeDetailExportMenu();
    });
    document.addEventListener('DOMContentLoaded', initializeEmployeeReconciliationDetail);
