    function canonicalEmployeeStore(value) {
      var store = String(value || '').trim();

        'sakura sushi & ramen house - dallas, tx 75247': '广州天河店',

    function dedupeEmployees(list) {
      var seenIds = Object.create(null);
      var seenProfiles = Object.create(null);
      return list.filter(function(employee) {
        var id = String(employee.employeeId || '').trim();
        var profileKey = [canonicalEmployeeStore(employee.store).toLowerCase(), String(employee.name || '').trim().toLowerCase(), String(employee.role || '').trim().toLowerCase()].join('|');
        if ((id && seenIds[id]) || seenProfiles[profileKey]) return false;
        if (id) seenIds[id] = true;
        seenProfiles[profileKey] = true;
        return true;
      });
    }

    function buildEmployeesFromRoster() {
      try {
        var raw = localStorage.getItem(EMPLOYEE_ROSTER_KEY);
        if (!raw) return DEFAULT_EMPLOYEES.slice();
        var list = JSON.parse(raw);
        if (!Array.isArray(list) || list.length === 0) return DEFAULT_EMPLOYEES.slice();
        var normalized = list
          .filter(function(e) { return e && String(e.id || '').trim() && String(e.name || '').trim(); })
          .map(function(e) {
            var role = String(e.role || '').trim();
            var tipType = String(e.tipType || 'deduct').trim();
            var baseTip = Number(e.baseTip) || 0;
            var tipRate = Number(e.tipRate);
            tipRate = Number.isFinite(tipRate) ? (tipRate > 1 ? tipRate / 100 : tipRate) : 0;
            return {
              employeeId: 'roster:' + String(e.id).trim(),
              name: String(e.name || '').trim(),
              store: canonicalEmployeeStore(e.store),
              role: role || 'Server',
              tipType: tipType === 'receive' ? 'receive' : 'deduct',
              baseTip: baseTip,
              tipRate: tipRate,
              tagClass: resolveTagClass({ role: role, tipType: tipType })
            };
          });
        return normalized.length ? dedupeEmployees(normalized) : DEFAULT_EMPLOYEES.slice();
      } catch (e) {
        return DEFAULT_EMPLOYEES.slice();
      }
    }

    function removeIndexMultiTag(e, btnEl) {
      e.stopPropagation();
      var selectId = btnEl && btnEl.dataset && btnEl.dataset.selectId;
      var value = btnEl && btnEl.dataset && btnEl.dataset.value;
      if (!selectId || value === undefined) return;
      var container = document.getElementById(selectId);
      var inputs = container ? container.querySelectorAll('input[type="checkbox"]') : [];
      for (var i = 0; i < inputs.length; i++) {
        if (inputs[i].value === value) { inputs[i].checked = false; break; }
      }
      refreshIndexMultiTags(selectId);
      if (selectId === 'roleFilter') updateEmployeeFilter();
      else renderDailySummaryList();
    }

    function refreshIndexMultiTags(selectId) {
      var container = document.getElementById(selectId);
      if (!container) return;
      var tagsEl = container.querySelector('.multi-select-tags');
      if (!tagsEl) return;
      var placeholder = (container.dataset && container.dataset.placeholder) || '请选择';
      var cbs = container.querySelectorAll('input[type="checkbox"]:checked:not([data-select-all])');
      tagsEl.innerHTML = '';
      if (cbs.length === 0) {
        tagsEl.innerHTML = '<span class="multi-select-placeholder">' + placeholder + '</span>';
        return;

      cbs.forEach(function(cb) {
        var tag = document.createElement('span');
        tag.className = 'multi-select-tag';
        tag.innerHTML = (cb.value || '') + ' <i data-select-id="' + selectId + '" data-value="' + (cb.value || '').replace(/"/g, '&quot;') + '" onclick="removeIndexMultiTag(event, this)">×</i>';
        tagsEl.appendChild(tag);
      });
    }

    function pad(n) { return n < 10 ? '0' + n : '' + n; }

    function money(v) { return (v < 0 ? '-' : '') + '$' + Math.abs(v).toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ','); }
    function moneySign(v) { return (v >= 0 ? '+' : '-') + '$' + Math.abs(v).toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ','); }
    function roundTip(v) { return +(Number(v) || 0).toFixed(2); }

    function seedRandom(seed) {
      let s = seed;
      return function() { s = (s * 16807 + 0) % 2147483647; return (s - 1) / 2147483646; };
    }

    function getPersonalSalesPctRule() {
      var rules = getLegacyRulesForCurrentStore();
      for (var i = 0; i < rules.length; i++) {
        var cfg = rules[i].deductConfig;
        if (cfg && cfg.personalSalesPct && window.TipOutPersonalSalesDeduct &&

    /** 演示期：由日期+姓名种子生成 Net Sales，隔离于他人销售额 */
    function genPersonalNetSales(emp, dateKey) {
      var seed = dateKey.split('-').reduce(function(a, b) { return a * 31 + parseInt(b, 10); }, 0) +
        emp.name.length * 13 + emp.name.charCodeAt(0) * 17;
      var rng = seedRandom(seed + 991);
      var dayOfWeek = new Date(dateKey + 'T00:00:00').getDay();
      var weekendBoost = (dayOfWeek === 5 || dayOfWeek === 6) ? 1.35 : 1.0;
      return roundTip((700 + rng() * 1600) * weekendBoost);
    }

    var dayPipelineCache = {};
    function invalidateDayPipelineCache() { dayPipelineCache = {}; }

    function ensureDayPipeline(dateKey) {
      if (dayPipelineCache[dateKey]) return dayPipelineCache[dateKey];
      var rule = getPersonalSalesPctRule();
      if (!rule || !window.TipAllocation || !TipAllocation.runLegacyDayPipeline) {
        dayPipelineCache[dateKey] = null;
        return null;
      }
      var dayEmps = employees.map(function(emp) {
        var seed = dateKey.split('-').reduce(function(a, b) { return a * 31 + parseInt(b, 10); }, 0) +
          emp.name.length * 7 + emp.name.charCodeAt(0);
        var rng = seedRandom(seed);
        var dayOfWeek = new Date(dateKey + 'T00:00:00').getDay();
        var weekendBoost = (dayOfWeek === 5 || dayOfWeek === 6) ? 1.35 : 1.0;
        var tipBefore;
        var deductCtx = getAllocationDeductorContext();
        var shouldDeduct = window.ruleData && ruleData.isEmployeeInDeductorContext
          ? ruleData.isEmployeeInDeductorContext(emp, deductCtx)
          : false;
        if (shouldDeduct || (emp.tipType || '') === 'deduct') {
          tipBefore = roundTip(emp.baseTip * (0.8 + rng() * 0.5) * weekendBoost);
        } else {
          var baseBefore = Number(emp.baseTip) > 0
            ? emp.baseTip * (0.8 + rng() * 0.5)
            : (5 + rng() * 12);
          tipBefore = roundTip(baseBefore * weekendBoost);
        }
        var attendance = (window.TipOutAttendance && TipOutAttendance.getDayStatus)
          ? TipOutAttendance.getDayStatus(emp.name, dateKey)
          : { hours: 0, clockStatus: '未打卡' };
        return {
          id: emp.name,
          name: emp.name,
          role: emp.role,
          tipBefore: tipBefore,
          tipRate: emp.tipRate || 0,
          salesAmount: genPersonalNetSales(emp, dateKey),
          hours: rule.clockin === 'noclock' ? 0 : attendance.hours
        };
      });
      var pipeline = TipAllocation.runLegacyDayPipeline(rule, dayEmps);
      dayPipelineCache[dateKey] = pipeline;
      return pipeline;
    }

    function attachAttendance(row, emp, dateKey) {
      var att = (window.TipOutAttendance && TipOutAttendance.getDayStatus)
        ? TipOutAttendance.getDayStatus(emp.name, dateKey)

    function genDailyTip(emp, dateKey) {
      var pipeline = ensureDayPipeline(dateKey);
      if (pipeline && pipeline.byName && pipeline.byName[emp.name]) {
        var row = pipeline.byName[emp.name];
        return attachAttendance({
          date: dateKey,
          before: row.before,
          deducted: row.deducted,
          received: row.received,
          after: row.after,
          due: row.due,
          shortfall: row.shortfall,
          salesAmount: row.salesAmount
        }, emp, dateKey);
      }

      const seed = dateKey.split('-').reduce((a, b) => a * 31 + parseInt(b), 0) + emp.name.length * 7 + emp.name.charCodeAt(0);
      const rng = seedRandom(seed);
      const dayOfWeek = new Date(dateKey + 'T00:00:00').getDay();
      const weekendBoost = (dayOfWeek === 5 || dayOfWeek === 6) ? 1.35 : 1.0;

      var before = 0;
      var deducted = 0;
      var received = 0;

      var deductCtx = getAllocationDeductorContext();
      var shouldDeduct = window.ruleData && ruleData.isEmployeeInDeductorContext
        ? ruleData.isEmployeeInDeductorContext(emp, deductCtx)
        : false;

      if (shouldDeduct) {
        before = roundTip(emp.baseTip * (0.8 + rng() * 0.5) * weekendBoost);
        deducted = roundTip(before * (emp.tipRate || 0));
        if (emp.role === 'Server') {
          received = roundTip((18 + rng() * 32) * weekendBoost);
        }
      } else {
        var baseBefore = Number(emp.baseTip) > 0
          ? emp.baseTip * (0.8 + rng() * 0.5)
          : (5 + rng() * 12);
        before = roundTip(baseBefore * weekendBoost);
        received = roundTip((25 + rng() * 45) * weekendBoost);
        deducted = 0;
      }

      var after = calcTipAfter(before, deducted, received);
      return attachAttendance({ date: dateKey, before: before, deducted: deducted, received: received, after: after }, emp, dateKey);
    }

    function getDateRange() {
      const s = document.getElementById('dateStart').value;
      const e = document.getElementById('dateEnd').value;
      if (!s || !e) return [];
      const start = new Date(s + 'T00:00:00');
      const end = new Date(e + 'T00:00:00');
      if (isNaN(start) || isNaN(end) || start > end) return [];
      const dates = [];
      const cur = new Date(start);
      while (cur <= end) {
        dates.push(formatDateKey(cur));
        cur.setDate(cur.getDate() + 1);
      }
      return dates;
    }

    function restoreSummaryUiState() {
      var saved = TipOutSummaryUi.readSummaryHistoryState(history.state);
      var signature = saved ? JSON.stringify(saved) : '__no_summary_state__';
      if (signature === restoredSummaryStateSignature) return false;
      restoredSummaryStateSignature = signature;
      if (!saved) return false;

      var queryView = new URLSearchParams(window.location.search).get('view');
      activeSummaryView = TipOutSummaryUi.resolveSummaryView(history.state, queryView);
      document.getElementById('dateStart').value = saved.dateStart || '';
      document.getElementById('dateEnd').value = saved.dateEnd || '';
      document.getElementById('storeSelect').value = saved.store || '';
      setSummaryFilterSelections('roleFilter', saved.roles);
      updateEmployeeFilter(true);
      setSummaryFilterSelections('employeeFilter', saved.employees);
      invalidateDayPipelineCache();
      syncSummaryViewUi();
      renderDailySummaryList();

      requestAnimationFrame(function() {
        window.scrollTo(0, Number(saved.scrollY) || 0);
        var row = null;
        if (saved.returnDate) row = document.querySelector('[data-date="' + saved.returnDate + '"]');
        if (!row && saved.returnEmployeeId) {
          var employeeRows = document.querySelectorAll('[data-employee-id]');
          for (var i = 0; i < employeeRows.length; i++) {
            if (employeeRows[i].dataset.employeeId === saved.returnEmployeeId) {
              row = employeeRows[i];
              break;
            }
          }
        }
        if (row) row.focus({ preventScroll: true });
      });
      return true;
    }

    function buildDailyDataset() {
      var allocatedDates = getAllocatedDates();
      var requiresAttendance = TipOutSummaryUi.resolveRequiresAttendance(getLegacyRulesForCurrentStore());
      return getDateRange().map(function(dateKey) {
        var employeeResults = getDailyEmployeeResults(dateKey);
        return {
          dateKey: dateKey,
          date: new Date(dateKey + 'T00:00:00'),
          allocated: allocatedDates.has(dateKey),
          requiresAttendance: requiresAttendance,
          employeeResults: employeeResults,
          total: TipOutSummaryUi.summarizeDailyResults(employeeResults)
        };
      });
    }

    function dailySummaryResultValue(allocated, value) {
      return allocated ? money(value) : '—';
    }

    function renderDateTaskList(dailyRows) {
      var container = document.getElementById('dailySummaryList');
      if (!container) return;
      container.innerHTML = dailyRows.map(function(row) {
        var dateKey = row.dateKey;
        var total = row.total;
        return '<tr class="tipout-daily-row" data-date="' + dateKey +
          '" tabindex="0" role="link" onclick="openDailyDetail(\'' + dateKey +
          '\')" onkeydown="activateDailySummaryRow(event, \'' + dateKey + '\')">' +
          '<td><span class="tipout-daily-date"><strong>' + formatTabLabel(row.date) +
          '</strong><small>' + dateKey + ' · ' + row.date.toLocaleDateString('zh-CN', { weekday: 'short' }) + '</small></span></td>' +
          '<td><span class="tipout-daily-status ' + (row.allocated ? 'allocated' : 'pending') + '">' +
          (row.allocated ? '已分配' : '待分配') + '</span></td>' +
          '<td>' + money(total.before) + '</td>' +
          '<td><strong class="tip-amount--deduct">' + dailySummaryResultValue(row.allocated, total.deducted) + '</strong></td>' +
          '<td><strong class="tip-amount--receive">' + dailySummaryResultValue(row.allocated, total.received) + '</strong></td>' +
          '<td><strong>' + dailySummaryResultValue(row.allocated, total.after) + '</strong></td>' +
          '<td class="tipout-detail-link">查看明细</td></tr>';
      }).join('');
    }

    function escapeSummaryHtml(value) {
      return String(value == null ? '' : value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
    }

    function employeeInitials(name) {
      var parts = String(name || '').trim().split(/\s+/).filter(Boolean);
      if (!parts.length) return '—';
      if (parts.length === 1) return Array.from(parts[0]).slice(0, 2).join('').toUpperCase();
      return (Array.from(parts[0])[0] + Array.from(parts[parts.length - 1])[0]).toUpperCase();
    }

    function renderEmployeeReconciliationList(dailyRows) {
      var container = document.getElementById('employeeReconciliationList');
      var empty = document.getElementById('employeeReconciliationEmpty');
      if (!container || !empty) return;
      var aggregates = TipOutSummaryUi.aggregateEmployeeDailyDatasets(dailyRows);
      employeeAggregateById = Object.create(null);
      aggregates.forEach(function(aggregate) { employeeAggregateById[aggregate.employeeId] = aggregate; });
      empty.hidden = aggregates.length > 0;
      container.closest('.tipout-table-wrap').hidden = aggregates.length === 0;
      container.innerHTML = aggregates.map(function(aggregate) {
        var statusClass = aggregate.status === '已完成' ? 'is-complete' : 'is-pending';
        var name = escapeSummaryHtml(aggregate.name);
        return '<tr class="tipout-employee-row" data-employee-id="' + escapeSummaryHtml(aggregate.employeeId) +
          '" tabindex="0" role="link" onclick="openEmployeeReconciliationDetail(this.dataset.employeeId)" ' +
          'onkeydown="activateEmployeeReconciliationRow(event)">' +
          '<td><span class="tipout-employee-identity"><span class="tipout-employee-avatar" aria-hidden="true">' +
          escapeSummaryHtml(employeeInitials(aggregate.name)) + '</span><strong>' + name + '</strong></span></td>' +
          '<td>' + escapeSummaryHtml(aggregate.role) + '</td>' +
          '<td><span class="tipout-shift-summary"><strong>' + aggregate.shifts + ' 个班次</strong><small>' +
          formatHoursDisplay(aggregate.hours) + ' h</small></span></td>' +
          '<td>' + money(aggregate.before) + '</td>' +
          '<td><strong class="tip-amount--deduct">-' + money(Math.abs(aggregate.deducted)) + '</strong></td>' +
          '<td><strong class="tip-amount--receive">' + moneySign(aggregate.received) + '</strong></td>' +
          '<td><strong>' + money(aggregate.after) + '</strong></td>' +
          '<td><span class="tipout-employee-status ' + statusClass + '"><span aria-hidden="true">●</span> ' + aggregate.status + '</span></td>' +
          '<td class="tipout-detail-link" aria-label="查看 ' + name + ' 对账明细">›</td></tr>';
      }).join('');
    }

    function renderSummaryViews() {
      var dailyRows = buildDailyDataset();
      var allocatedDates = getAllocatedDates();
      renderSummaryOverview(dailyRows, allocatedDates);
      if (activeSummaryView === 'employee') {
        document.getElementById('dailySummaryList').innerHTML = '';
        renderEmployeeReconciliationList(dailyRows);
      } else {
        document.getElementById('employeeReconciliationList').innerHTML = '';
        document.getElementById('employeeReconciliationEmpty').hidden = true;
        renderDateTaskList(dailyRows);
      }
      updateAllocateButton();
    }

    function renderDailySummaryList() { renderSummaryViews(); }

    function updateAllocateButton() {
      var dates = getDateRange();
      var allocatedDates = getAllocatedDates();
      var hasAllocated = dates.some(function(k) { return allocatedDates.has(k); });
      var btn = document.getElementById('allocateBtn');
      if (btn) btn.textContent = hasAllocated ? '重新分配小费' : '分配小费';
    }

    function openSummaryRuleEntry() {
      var rules = [];
      try {
        rules = window.ruleData && ruleData.getRules ? ruleData.getRules() : [];
      } catch (e) {
        rules = [];
      }
      var hasRules = Array.isArray(rules) && rules.length > 0;
      window.location.href = hasRules ? 'rules.html' : 'rule-add.html?poolKind=tip';
    }

    function hasTipRules() {
      return window.ruleData && ruleData.getRules && ruleData.getRules().length > 0;
    }

    function doAllocateTips() {
      if (!hasTipRules()) {
        showNotification('请先创建小费分配规则', 'error');
        setTimeout(function() { window.location.href = 'rules.html'; }, 1500);
        return;
      }
      invalidateDayPipelineCache();
      var dates = getDateRange();
      var allocatedDates = getAllocatedDates();
      dates.forEach(function(k) { allocatedDates.add(k); });
      saveAllocatedDates(allocatedDates);
      renderDailySummaryList();
      if (window.TipOutPayrollBridge && TipOutPayrollBridge.syncAfterAllocation) {
        var storeSel = document.getElementById('storeSelect');
        var storeVal = storeSel ? (storeSel.value || '').trim() : '';
        var ds = document.getElementById('dateStart');
        var de = document.getElementById('dateEnd');
        TipOutPayrollBridge.syncAfterAllocation(
          storeVal,
          ds ? ds.value : '',
          de ? de.value : ''
        );
      }
      showNotification('小费分配完成，已同步至薪资管理 Tips/SVCW', 'success');
    }

    function doCancelAllocate() {
      invalidateDayPipelineCache();
      var dates = getDateRange();
      var allocatedDates = getAllocatedDates();
      dates.forEach(function(k) { allocatedDates.delete(k); });
      saveAllocatedDates(allocatedDates);
      renderDailySummaryList();
      if (window.TipOutPayrollBridge && TipOutPayrollBridge.syncAfterAllocation) {
        var storeSel = document.getElementById('storeSelect');
        var storeVal = storeSel ? (storeSel.value || '').trim() : '';
        var ds = document.getElementById('dateStart');
        var de = document.getElementById('dateEnd');
        TipOutPayrollBridge.syncAfterAllocation(
          storeVal,
          ds ? ds.value : '',
          de ? de.value : ''
        );
      }
      showNotification('已取消分配，状态已恢复为待分配', 'info');
    }

    function getFilteredEmployees() {
      var roles = getSelectedRoles();
      var empNames = getSelectedEmployees();
      var list = employees;
      var storeSelect = document.getElementById('storeSelect');
      var selectedStore = canonicalEmployeeStore(storeSelect ? storeSelect.value : '');
      if (selectedStore) list = list.filter(function(e) { return !e.store || canonicalEmployeeStore(e.store) === selectedStore; });
      if (roles.length > 0) list = list.filter(function(e) { return roles.indexOf(e.role) >= 0; });
      if (empNames.length > 0) list = list.filter(function(e) { return empNames.indexOf(e.name) >= 0; });
      return list;
    }

    function toggleFold(empId) {
      const section = document.getElementById(empId);
      const detail = section.querySelector('.employee-detail');
      const btn = section.querySelector('.fold-btn span');
      if (detail.style.maxHeight === '0px') { detail.style.maxHeight = '2000px'; btn.textContent = '收起'; }
      else { detail.style.maxHeight = '0px'; btn.textContent = '展开'; }
    }

    function initStoreSelect() {
      var sel = document.getElementById('storeSelect');
      if (!sel) return;
      var rules = window.ruleData && ruleData.getRules ? ruleData.getRules() : [];
      var seen = {};
      var stores = [];
      rules.forEach(function(r) {
        var s = (r.store || '').trim();
        if (s && !seen[s]) { seen[s] = 1; stores.push(s); }
      });
      sel.innerHTML = '<option value="">请选择门店</option>' + stores.map(function(s) {
        return '<option value="' + s + '">' + s + '</option>';
      }).join('');
      if (stores.length > 0 && !sel.value) sel.value = stores[0];
    }

    document.addEventListener('change', function(e) {
      var target = e.target;
      var ms = target && target.closest && target.closest('.filter-bar .multi-select');
      if (!ms) return;
      var id = ms.id;
      if (target.dataset && target.dataset.selectAll !== undefined) {
        toggleSelectAll(id, target.checked);
        return;
      }
      refreshIndexMultiTags(id);
      syncSelectAllState(id);
      if (id === 'roleFilter') updateEmployeeFilter();
      else if (id === 'employeeFilter') renderDailySummaryList();
    });
    document.addEventListener('click', function(e) {
      if (!e.target.closest || !e.target.closest('.filter-bar .multi-select')) {
        document.querySelectorAll('.filter-bar .multi-select.open').forEach(function(ms) { ms.classList.remove('open'); });
      }
    });

    function initializeSummaryPage() {
      initStoreSelect();
      activeSummaryView = TipOutSummaryUi.resolveSummaryView(
        history.state,
        new URLSearchParams(window.location.search).get('view')
      );
      syncSummaryViewUi();
      if (!restoreSummaryUiState()) {
        updateEmployeeFilter();
        renderDailySummaryList();
      }
    }

    document.addEventListener('DOMContentLoaded', initializeSummaryPage);
    window.addEventListener('pageshow', function() {
      restoreSummaryUiState();
    });
