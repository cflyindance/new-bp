function isStoreSummaryRoute(){var p=new URLSearchParams(window.location.search);return p.get('screen')==='store'||p.get('view')==='employee';}
function isAllocationDashboard(){return !isStoreSummaryRoute();}
function dashboardRange(){return {startDate:document.getElementById('dateStart').value,endDate:document.getElementById('dateEnd').value};}
function dashboardAllowedStores(){
 var options=window.TipOutGlobalScopeFilter&&TipOutGlobalScopeFilter.listScopedStoreOptions?TipOutGlobalScopeFilter.listScopedStoreOptions():[];
 var rules=ruleData.getRules(),seen={};
 return options.map(function(o){var label=o.labelZh||o.labelEn||o.value,canon=canonicalEmployeeStore(label);var rule=rules.find(function(r){return canonicalEmployeeStore(r.store)===canon||canonicalEmployeeStore(r.store)===canonicalEmployeeStore(o.labelEn);});return {id:o.value,label:label,store:rule?rule.store:label};}).filter(function(s){var k=canonicalEmployeeStore(s.store);if(!s.id||seen[k])return false;seen[k]=true;return true;});
}
function dashboardScopeKey(){var range=dashboardRange();return JSON.stringify([dashboardAllowedStores().map(function(s){return [s.id,s.store];}),range.startDate,range.endDate]);}

}

function assertSummaryAutoActive(){if(summaryAutoDisposed)throw new Error('页面已离开，停止自动分配');if(summaryAutoScope&&summaryAutoScope.key!==dashboardScopeKey())throw new Error('可访问门店或日期范围已变化，请刷新看板');}
function summaryAutoExisting(store,date){var state=TipOutDateState.inspect(store,date);if(state.error)throw new Error(state.error);if(state.payoutStatus==='paid')return {status:'skipped',message:'已发放，保留原结果'};if(state.snapshot||TipOutAllocationResults.isAllocated(store,date))return {status:'skipped',message:'已分配，保留原结果'};return null;}
async function dashboardInspectStores(){
 var allowed=dashboardAllowedStores(),range=dashboardRange();dashboardStores=[];dashboardIssues=[];
 for(var i=0;i<allowed.length;i++){
  assertSummaryAutoActive();var s=Object.assign({},allowed[i]);s.ruleCount=ruleData.getRulesForStore(s.store).length;s.days=[];
  document.getElementById('dashboardCurrent').textContent='正在检查 '+s.label+' · '+(i+1)+' / '+allowed.length+' 家门店';
  await new Promise(function(resolve){window.setTimeout(resolve,20);});assertSummaryAutoActive();
  if(s.ruleCount){try{var rows=readAllocationScopeRows({store:s.store,startDate:range.startDate,endDate:range.endDate});s.days=rows.map(function(r){var state=TipOutDateState.inspect(s.store,r.dateKey);return {date:r.dateKey,allocatable:!!(r.businessStatusResult&&r.businessStatusResult.allocatable),allocated:!!(state.snapshot||r.allocated),paid:state.payoutStatus==='paid'};});}catch(error){s.error=error.message||'读取门店日期失败';dashboardIssues.push({store:s.store,storeLabel:s.label,message:s.error});}}
  dashboardStores.push(s);
 }
}
async function executeSummaryAutoDate(date,item){
 await new Promise(function(resolve){window.setTimeout(resolve,60);});assertSummaryAutoActive();var store=item.store;
 document.getElementById('storeSelect').value=store;var existing=summaryAutoExisting(store,date);if(existing)return existing;
 var rows=readAllocationScopeRows({store:store,startDate:date,endDate:date});if(!rows[0]||!rows[0].businessStatusResult)throw new Error('无法读取当天营业数据');if(!rows[0].businessStatusResult.allocatable)return {status:'skipped',message:'未营业，无需分配'};
 var rules=getRulesForCurrentStore();if(!rules.length)throw new Error('当前门店没有可用的小费分配规则');if(rules.some(function(r){return r.clockin==='noclock';}))throw new Error('需在当天明细补全手动工时后分配');
 var snapshot=window.prepareTipoutQuickSnapshot(store,date);
 try{await TipOutAllocationResults.commit(snapshot,{onlyIfUnallocated:true,assertActive:assertSummaryAutoActive});}catch(error){assertSummaryAutoActive();var concurrent=summaryAutoExisting(store,date);if(concurrent)return concurrent;throw error;}
 var message='系统分配已保存';try{TipOutDateState.notify(store,date);}catch(_){message+='；汇总通知失败，请刷新查看';}
 try{if(window.TipOutPayrollBridge)TipOutPayrollBridge.syncAfterAllocation(store,date,date);}catch(_){message+='；薪资同步失败，请更新员工数据';}
 return {status:'success',message:message};
}
function dashboardRefreshSavedDays(){dashboardStores.forEach(function(s){s.days.forEach(function(d){var state=TipOutDateState.inspect(s.store,d.date);d.allocated=!!(state.snapshot||TipOutAllocationResults.isAllocated(s.store,d.date));d.paid=state.payoutStatus==='paid';});});}

function renderSummaryAutoProgress(queue){if(summaryAutoDisposed)return;dashboardLastItems=queue.items;if(!queue.running){dashboardCompleted=true;dashboardSetBusy(false);dashboardSaveReport();}renderDashboardProgress();}
async function startSummaryAutoAllocation(){
 if(summaryAutoDisposed||dashboardBusy||!isAllocationDashboard()||!validateSummaryDateRange(false))return;
 summaryAutoScope=Object.assign(dashboardRange(),{key:dashboardScopeKey()});summaryAutoQueue=null;dashboardLastItems=[];dashboardCompleted=false;dashboardInspecting=true;dashboardSetBusy(true);renderDashboardProgress();
 try{await dashboardInspectStores();assertSummaryAutoActive();var plan=TipOutDashboardModel.plan(dashboardStores),report=dashboardReadReport(summaryAutoScope.key);dashboardInspecting=false;
  var failedKeys=report?report.items.filter(function(i){return i.status==='failed';}).map(function(i){return TipOutDashboardModel.key(i.store,i.date);}):[];
  if(report&&plan.every(function(i){return failedKeys.indexOf(TipOutDashboardModel.key(i.store,i.date))>=0;})){dashboardLastItems=report.items.map(function(i){if(i.status==='failed'&&summaryAutoExisting(i.store,i.date))return Object.assign({},i,{status:'skipped',message:'已另行分配，保留原结果'});return i;});dashboardCompleted=true;dashboardSetBusy(false);renderDashboardProgress();return;}
  if(!plan.length){dashboardCompleted=true;dashboardSetBusy(false);renderDashboardProgress();dashboardSaveReport();return;}
  summaryAutoQueue=window.TipOutAutoAllocation.create(plan,executeSummaryAutoDate,renderSummaryAutoProgress);await summaryAutoQueue.run();
 }catch(error){if(!summaryAutoDisposed){dashboardInspecting=false;dashboardIssues.push({message:error.message||'自动分配无法启动'});dashboardSetBusy(false);renderDashboardProgress();}}
}
