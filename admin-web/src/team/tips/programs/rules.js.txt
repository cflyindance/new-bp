function initStoreFilter() {
      var sel = document.getElementById('storeFilter');
      if (!sel) return;
      var rules = ruleData.getRules();
      var seen = {};
      var stores = [];
      rules.forEach(function(r) {
        var s = (r.store || '').trim();
        if (s && !seen[s]) { seen[s] = 1; stores.push(s); }
      });
      sel.innerHTML = '<option value="">全部门店</option>' + stores.map(function(s) {
        return '<option value="' + s + '">' + s + '</option>';
      }).join('');
    }

    function escapeHtml(value) {
      return String(value == null ? '' : value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
    }

    function renderRuleMetrics(rules) {
      var list = rules || [];
      document.getElementById('ruleCount').textContent = String(list.length);
      document.getElementById('tipPoolCount').textContent = String(list.filter(function(rule) {
        return rule.poolKind !== 'surcharge';
      }).length);
      document.getElementById('surchargePoolCount').textContent = String(list.filter(function(rule) {
        return rule.poolKind === 'surcharge';
      }).length);
    }

    function clearRuleNameSearch() {
      var search = document.getElementById('ruleNameSearch');
      if (!search) return;
      search.value = '';
      renderRulesTable();
      search.focus();
    }

    function renderRulesTable() {
      var rules = ruleData.getRules();
      var search = document.getElementById('ruleNameSearch');
      var keyword = String(search ? search.value : '').trim().toLocaleLowerCase();
      var clearSearch = document.getElementById('clearRuleNameSearch');
      if (clearSearch) clearSearch.hidden = !search || !search.value;
      if (keyword) rules = rules.filter(function(rule) {
        return String(rule.ruleName || '').toLocaleLowerCase().includes(keyword);
      });
      var storeFilter = document.getElementById('storeFilter');
      var filterStore = storeFilter ? storeFilter.value : '';
      if (filterStore) {
        rules = rules.filter(function(r) { return r.store === filterStore; });
      }
      var poolKindFilter = document.getElementById('poolKindFilter');
      var filterKind = poolKindFilter ? poolKindFilter.value : '';
      if (filterKind) {
        rules = rules.filter(function(rule) { return (rule.poolKind === 'surcharge' ? 'surcharge' : 'tip') === filterKind; });
      }
      renderRuleMetrics(rules);
      var tableBody = document.getElementById('rulesTableBody');
      tableBody.innerHTML = '';
      if (rules.length === 0) {
        tableBody.innerHTML = '<tr><td colspan="4"><div class="tipout-rule-empty" role="status">' + (keyword || filterStore || filterKind ? '暂无符合条件的规则' : '暂无规则，请点击「新增规则」创建') + '</div></td></tr>';
        return;
      }
      rules.forEach(function(rule) {
        var referenceIssues = window.TipOutRosterDirectory && TipOutRosterDirectory.validateRuleReferences
          ? TipOutRosterDirectory.validateRuleReferences(rule, rule.store)
          : [];
        var record = document.createElement('tr');
        record.className = 'tipout-rule-record';
        record.setAttribute('data-rule-id', rule.id);
        var editUrl = 'rule-add.html?mode=edit&id=' + rule.id;
        if (rule.poolKind === 'surcharge') editUrl += '&poolKind=surcharge';
        var poolKindLabel = ruleData.getRulePoolKindLabel(rule);
        var poolKindTagClass = ruleData.getRulePoolKindTagClass(rule);
        var menuId = 'ruleActions-' + rule.id;
        record.innerHTML =
          '<td>' +
            '<div class="tipout-rule-identity">' +
              '<strong class="tipout-rule-name">' + escapeHtml(rule.ruleName || '') + '</strong>' +
              '<span class="status-chip">' + (rule.enabled === false ? '已禁用' : '已启用') + '</span>' +
              (referenceIssues.length ? '<span class="status-chip status-warning">引用失效</span>' : '') +
            '</div>' +
          '</td>' +
          '<td><span class="tipout-rule-store" title="' + escapeHtml(rule.store || '') + '">' + escapeHtml(rule.store || '') + '</span></td>' +
          '<td>' +
            '<div class="tipout-rule-pool-summary">' +
              '<span class="' + escapeHtml(poolKindTagClass) + '">' + escapeHtml(poolKindLabel) + '</span>' +
            '</div>' +
          '</td>' +
          '<td><div class="tipout-rule-record-actions">' +
            '<a class="btn btn-primary tipout-rule-edit" href="' + editUrl + '">编辑</a>' +
            '<div class="tipout-rule-actions">' +
              '<button type="button" class="btn tipout-rule-more" aria-haspopup="menu" aria-expanded="false" aria-controls="' + menuId + '" onclick="toggleRuleActions(this)">更多操作</button>' +
              '<div class="tipout-rule-actions-menu" id="' + menuId + '" role="menu" hidden>' +
                '<button type="button" role="menuitem" onclick="toggleRuleEnabled(this)">' + (rule.enabled === false ? '启用规则' : '禁用规则') + '</button>' +
                '<button type="button" role="menuitem" onclick="copyRule(this)">复制规则</button>' +
                '<button type="button" role="menuitem" class="danger" onclick="deleteRule(this)">删除规则</button>' +
              '</div>' +
            '</div>' +
          '</div></td>';
        tableBody.appendChild(record);
      });
    }

    function closeRuleActions(exceptButton) {
      document.querySelectorAll('.tipout-rule-more[aria-expanded="true"]').forEach(function(button) {
        if (button === exceptButton) return;
        button.setAttribute('aria-expanded', 'false');
        var menu = document.getElementById(button.getAttribute('aria-controls'));
        if (menu) menu.hidden = true;
      });
    }

    function toggleRuleActions(button) {
      var willOpen = button.getAttribute('aria-expanded') !== 'true';
      closeRuleActions(button);
      button.setAttribute('aria-expanded', String(willOpen));
      var menu = document.getElementById(button.getAttribute('aria-controls'));
      if (menu) {
        menu.hidden = !willOpen;
        if (willOpen) {
          var firstAction = menu.querySelector('[role="menuitem"]');
          if (firstAction) firstAction.focus();
        }
      }
    }

    function deleteRule(el) {
      if(ruleStatusBusy){showNotification('正在重算，请稍后操作');return;}
      confirmAction('确定要删除该规则吗？此操作不可恢复。', function() {
        var row = el.closest('[data-rule-id]');
        var id = row.getAttribute('data-rule-id');
        ruleData.deleteRuleById(id);
        row.style.opacity = '0';
        row.style.transition = 'opacity 0.3s';
        setTimeout(function() { renderRulesTable(); }, 300);
        showNotification('规则已删除');
      }, { title: '删除规则', okText: '删除', okClass: 'btn btn-danger' });
    }

    function openAddRuleModal() { location.href = buildRuleAddUrl(); }

    function buildRuleAddUrl(poolType) {
      var params = [];
      if (poolType) params.push('poolKind=' + encodeURIComponent(poolType));
      try {
        var embedded = new URLSearchParams(window.location.search).get('embedded');
        if (embedded) params.push('embedded=' + encodeURIComponent(embedded));
      } catch (e) {}
      return params.length ? ('rule-add.html?' + params.join('&')) : 'rule-add.html';
    }

    function copyRule(el) {
      if(ruleStatusBusy){showNotification('正在重算，请稍后操作');return;}
      var row = el.closest('[data-rule-id]');
      var id = row.getAttribute('data-rule-id');
      var rule = ruleData.getRuleById(id);
      if (!rule) return;
      var rules = ruleData.getRules();
      var newRule = JSON.parse(JSON.stringify(rule));
      newRule.id = ruleData.getNextRuleId();
      newRule.ruleName = (rule.ruleName || '') + ' (副本)';
      rules.push(newRule);
      ruleData.saveRules(rules);
      renderRulesTable();
      showNotification('规则复制成功');
    }

    document.addEventListener('click', function(event) {
      if (!event.target.closest('.tipout-rule-actions')) closeRuleActions();
    });

    document.addEventListener('keydown', function(event) {
      if (event.key !== 'Escape') return;
      var openButton = document.querySelector('.tipout-rule-more[aria-expanded="true"]');
      if (!openButton) return;
      closeRuleActions();
      openButton.focus();
    });

    try {
      var demoRuleCount = window.TipOutDemoRules ? window.TipOutDemoRules.ensure() : 0;
      if (demoRuleCount) showNotification('已生成 ' + demoRuleCount + ' 条演示规则，可编辑或删除；不会修改历史分配快照');
    } catch (error) {
      showNotification('演示规则生成失败，请检查本地存储后重试');
    }
    initStoreFilter();

var ruleStatusJobKey = 'tipout_rule_status_recalculation_v1';
function ruleStatusActive() { if (window.tipoutIsPageActive && !window.tipoutIsPageActive()) throw new Error('已离开规则页面，待返回后重试'); }
function saveRuleStatusJob(job) { localStorage.setItem(ruleStatusJobKey, JSON.stringify(job)); }
function showRuleStatusJob(job, running) {
  var box=document.getElementById('ruleStatusReport');
  if(!box){box=document.createElement('section');box.id='ruleStatusReport';box.className='rule-status-report';box.setAttribute('aria-live','polite');document.getElementById('rulesTableBody').closest('table').before(box);}
  box.replaceChildren();var title=document.createElement('strong');title.textContent=running?'正在重新分配小费':'规则禁用处理结果';box.appendChild(title);
  job.dates.forEach(function(item){var p=document.createElement('p');p.textContent=item.date+' · '+({pending:'等待处理',done:'已重算',paid:'已发放，保留结果',failed:'处理失败'}[item.status]||item.status)+(item.error?'：'+item.error:'');box.appendChild(p);});
  if(!running && job.dates.some(function(x){return x.status==='failed'||x.status==='pending';})){var retry=document.createElement('button');retry.className='btn btn-primary';retry.textContent='重试未完成日期';retry.addEventListener('click',function(){confirmAction('将按当前规则重新系统分配未完成日期，并覆盖这些日期当前的手动调整；已发放日期仍保留。',function(){job.dates.forEach(function(item){if(item.status==='failed'||item.status==='pending'){item.fingerprint=JSON.stringify(TipOutAllocationResults.read(job.store,item.date));item.status='pending';}});saveRuleStatusJob(job);runRuleStatusJob(job);},{title:'重试日期重算',okText:'确认重试'});});box.appendChild(retry);}
}
async function runRuleStatusJob(job) {
  if(ruleStatusBusy)return;ruleStatusBusy=true;
  try {
    for(var item of job.dates){
      if(item.status==='done'||item.status==='paid')continue;
      try {
        ruleStatusActive();var rule=ruleData.getRuleById(job.ruleId);if(!rule||rule.enabled!==false)throw new Error('规则状态已变化，无法继续本次重算');
        var state=TipOutDateState.inspect(job.store,item.date);if(state.locked){if(state.payoutStatus==='paid'){item.status='paid';item.error='';saveRuleStatusJob(job);continue;}throw new Error(state.error||'日期已锁定');}
        var current=TipOutAllocationResults.read(job.store,item.date);
        if(JSON.stringify(current)!==item.fingerprint)throw new Error('分配结果已被其他操作修改，请核对后重新发起');
        var config=JSON.stringify(ruleData.getRules());
        function assertCurrent(){ruleStatusActive();if(JSON.stringify(ruleData.getRules())!==config)throw new Error('规则配置已变化，请重试');}
        if(ruleData.getRulesForStore(job.store).length){
          var snapshot=await window.prepareTipoutQuickSnapshot(job.store,item.date,true);
          assertCurrent();await TipOutAllocationResults.commit(snapshot,{expectedSnapshotFingerprint:item.fingerprint,assertActive:assertCurrent});
        }else{await TipOutAllocationResults.cancel(job.store,item.date,{expectedSnapshotFingerprint:item.fingerprint,assertActive:assertCurrent});}
        item.status='done';item.error='';
        try { if (window.TipOutPayrollBridge) window.TipOutPayrollBridge.syncAfterAllocation(job.store,item.date,item.date); } catch(syncError) { item.error='分配已更新，薪资同步失败：'+syncError.message; }
      }catch(error){item.status='failed';item.error=error.message||String(error);}
      saveRuleStatusJob(job);showRuleStatusJob(job,true);
      if(window.tipoutIsPageActive&&!window.tipoutIsPageActive())break;
    }
  }finally{ruleStatusBusy=false;saveRuleStatusJob(job);showRuleStatusJob(job,false);renderRulesTable();}
}
function toggleRuleEnabled(el){
  if(ruleStatusBusy){showNotification('正在重算，请稍后操作');return;}
  var previousJob=JSON.parse(localStorage.getItem(ruleStatusJobKey)||'null');
  if(previousJob&&previousJob.dates.some(function(x){return x.status==='pending'||x.status==='failed';})){showNotification('请先完成上次规则禁用的日期重算');showRuleStatusJob(previousJob,false);return;}
  var id=el.closest('[data-rule-id]').getAttribute('data-rule-id'),rule=ruleData.getRuleById(id);if(!rule)return;
  var disabling=rule.enabled!==false,dates=[],paid=0,manual=0;
  try{if(disabling){var results=JSON.parse(localStorage.getItem('tipout_allocation_results_v1')||'{}');Object.values(results).forEach(function(snapshot){if(snapshot.store!==rule.store||!Array.isArray(snapshot.pools)||!snapshot.pools.some(function(p){return String(p.ruleId)===String(id);}))return;var state=TipOutDateState.inspect(rule.store,snapshot.dateKey);if(state.payoutStatus==='paid'){paid++;return;}if(snapshot.allocationSource==='manual')manual++;dates.push({date:snapshot.dateKey,status:'pending',fingerprint:JSON.stringify(TipOutAllocationResults.read(rule.store,snapshot.dateKey))});});}}catch(error){showNotification('无法读取分配结果：'+error.message);return;}
  var fingerprint=JSON.stringify(rule);
  confirmAction(disabling?'禁用后不再参与分配。将重新系统分配 '+dates.length+' 个待发放日期，覆盖这些日期的手动调整（其中 '+manual+' 个手动分配日期）。保留 '+paid+' 个已发放日期。'+(dates.length?'日期：'+dates.map(function(x){return x.date;}).join('、'):''):'启用后恢复参与后续分配，历史结果保持不变。',function(){
    try{var rules=ruleData.getRules(),target=rules.find(function(r){return String(r.id)===String(id);});if(!target||JSON.stringify(target)!==fingerprint)throw new Error('规则已变化，请重新操作');target.enabled=!disabling;ruleData.saveRules(rules);renderRulesTable();if(disabling&&dates.length){var job={ruleId:id,store:rule.store,dates:dates};saveRuleStatusJob(job);runRuleStatusJob(job);}else showNotification(disabling?'规则已禁用':'规则已启用');}catch(error){showNotification(error.message);}
  },{title:disabling?'禁用并重算待发放日期':'启用规则',okText:disabling?'禁用并重算':'启用'});
}
