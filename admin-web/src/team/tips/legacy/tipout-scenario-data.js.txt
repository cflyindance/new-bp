/** Read-only demo facts. No clock, DOM, roster mutation, or business-storage writes. */
(function(root) {
  'use strict';
  var VERSION = 'tipout-scenarios-v1';
  var DAY = 86400000;
  function required(value, label) {
    if (typeof value !== 'string' || !value.trim()) throw new Error(label + ' is required');
    return value.trim();
  }
  function dateValue(key) {
    if (!/^\d{4}-\d{2}-\d{2}$/.test(String(key))) throw new Error('Invalid scenario date');
    var value = Date.parse(key + 'T00:00:00Z');
    if (!isFinite(value) || new Date(value).toISOString().slice(0, 10) !== key) throw new Error('Invalid scenario date');
    return value;
  }
  function hash(value) {
    var result = 2166136261;
    for (var i = 0; i < value.length; i++) { result ^= value.charCodeAt(i); result = Math.imul(result, 16777619); }
    return result >>> 0;
  }
  function monthDates(todayKey) {
    var end = dateValue(todayKey), today = new Date(end);
    var previous = new Date(end);
    previous.setUTCDate(1);
    previous.setUTCMonth(previous.getUTCMonth() - 1);
    var last = new Date(end);
    last.setUTCDate(0);
    previous.setUTCDate(Math.min(today.getUTCDate(), last.getUTCDate()));
    var dates = [];
    for (var cursor = previous.getTime(); cursor <= end; cursor += DAY) dates.push(new Date(cursor).toISOString().slice(0, 10));
    return dates;
  }
  function time(minutes) { return String(Math.floor(minutes / 60)).padStart(2, '0') + ':' + String(minutes % 60).padStart(2, '0'); }
  function chicagoOffset(dateKey) {
    var label = new Intl.DateTimeFormat('en-US', { timeZone: 'America/Chicago', timeZoneName: 'shortOffset' })
      .formatToParts(new Date(dateKey + 'T12:00:00Z')).filter(function(part) { return part.type === 'timeZoneName'; })[0].value;
    var hours = Number((label.match(/GMT([+-]\d+)/) || [])[1]);
    if (!Number.isFinite(hours)) throw new Error('Cannot resolve store time zone');
    return (hours < 0 ? '-' : '+') + String(Math.abs(hours)).padStart(2, '0') + ':00';
  }
  function session(start, hours) { return { clockIn: time(start), clockOut: time(start + hours * 60), durationHours: hours, status: 'complete' }; }
  function datedSession(item, dateKey) {
    var copy = Object.assign({}, item);
    var offset = chicagoOffset(dateKey);
    copy.clockInAt = dateKey + 'T' + item.clockIn + ':00' + offset;
    if (item.clockOut) copy.clockOutAt = dateKey + 'T' + item.clockOut + ':00' + offset;
    return copy;
  }
  function attendance(clock, variant) {
    if (!clock) return { hours: 0, originalHours: 0, effectiveHours: 0, clockStatus: '未打卡', punchSessions: [], adjustment: null };
    var sessions, original = 8, effective = 8, status = '已打卡';
    if (variant === 1) sessions = [session(480, 4), session(1020, 4)];
    else if (variant === 2) {
      original = effective = 4; status = '打卡异常';
      sessions = [session(480, 4), { clockIn: '17:00', clockOut: null, durationHours: null, status: 'missing-clock-out' }];
    } else sessions = [session(480, 8)];
    if (variant === 3) effective = 10;
    return { hours: effective, originalHours: original, effectiveHours: effective, clockStatus: status, punchSessions: sessions,
      adjustment: variant === 3 ? { source: 'POS', fromHours: 8, toHours: 10 } : null };
  }
  function fact(options) {
    var storeId = required(options.storeId, 'storeId');
    var employee = options.employee || {}, employeeId = required(employee.id, 'employeeId');
    var calendarDay = dateValue(options.dateKey) / DAY;
    var identity = JSON.stringify([VERSION, storeId, employeeId]);
    var seed = hash(identity), combination = ((calendarDay + seed) % 8 + 8) % 8;
    var key = JSON.stringify([VERSION, storeId, employeeId, options.dateKey]);
    var shortFixture=['上海陆家嘴店','Golden Dragon Chinese Kitchen - Dallas, TX 75231'].indexOf(storeId)>=0&&employeeId==='tipout-daily-shortfall-source';
    // Opt-in isolated acceptance identities; never added to the ordinary roster automatically.
    var shareFixture=['上海陆家嘴店','Golden Dragon Chinese Kitchen - Dallas, TX 75231'].indexOf(storeId)>=0&&['tipout-share-test-a','tipout-share-test-b','tipout-share-test-receiver'].indexOf(employeeId)>=0;
    var dailyFixture=['上海陆家嘴店','Golden Dragon Chinese Kitchen - Dallas, TX 75231'].indexOf(storeId)>=0&&['tipout-daily-excess-source','tipout-daily-excess-kitchen','tipout-daily-excess-busser','tipout-daily-shortfall-source'].indexOf(employeeId)>=0;
    if(storeId==='【演示】贡献超计划验收门店'||dailyFixture||shareFixture){
      var donor=employeeId==='funding-source'||employeeId==='tipout-daily-excess-source'||shortFixture||(shareFixture&&employeeId!=='tipout-share-test-receiver');
      var tip=shareFixture?(employeeId==='tipout-share-test-a'?20:employeeId==='tipout-share-test-b'?1000:0):donor?(shortFixture?5:20):0;
      var sales=shareFixture?(employeeId==='tipout-share-test-a'?200:employeeId==='tipout-share-test-b'?800:0):donor?100:0;
      var punch=attendance(shareFixture||dailyFixture||['funding-source','funding-kitchen','funding-busser'].indexOf(employeeId)>=0,0);
      punch.punchSessions=punch.punchSessions.map(function(item){return datedSession(item,options.dateKey);});
      return {version:VERSION,key:key,storeId:storeId,employeeId:employeeId,name:String(employee.name||''),role:String(employee.role||''),dateKey:options.dateKey,scenarioId:shareFixture?'CONTRIBUTOR-SHARES':shortFixture?'FUNDING-SHORTFALL':'FUNDING-EXCESS',contributionScenario:shareFixture?'隔离验收：贡献应扣100，A应扣40实扣20，B扣60，老板补20':shortFixture?'隔离演示：计划10、贡献5、老板补5，后厨6、打杂4':'隔离演示：计划10、贡献20，后厨12、打杂8',attendance:punch,
        orders:donor?[{id:key+':order:0',employeeId:employeeId,businessDate:options.dateKey,salesAmount:sales,tipAmount:tip,surcharge:0,orderType:'dine_in',area:'main',paymentMethod:'card',localTime:'11:30',completedPaymentAt:options.dateKey+'T11:30:00'+chicagoOffset(options.dateKey),lines:[{productId:'demo-main',productLine:'POS',category:'主菜',amount:sales}]}]:[],
        reportedTipEvents:[],salesAmount:sales,orderTips:tip,reportedTips:0,originalTips:tip,surcharge:0};
    }
    var dailySeed = hash(key), hasOrders = combination % 4 < 2, hasTips = combination % 2 === 0;
    var salesCents = hasOrders ? 12000 + dailySeed % 50001 : 0;
    var tipCents = hasTips ? 1 + hash(key + ':tip') % 6000 : 0;
    // Preserve B01–B08 coverage; enrich clocked sales/tip days with donor cases.
    var contributionScenario = null;
    if (combination === 0 && (employee.role === 'Server' || employee.role === 'Bartender')) {
      var insufficient = options.fundingShortfall === true;
      salesCents = employee.role === 'Server' ? 100000 : 60000;
      tipCents = insufficient ? 100 : Math.round(salesCents * 0.2);
      contributionScenario = insufficient ? '贡献入池封顶' : (employee.role === 'Server' ? '服务员贡献入池' : '调酒师贡献并接收');
    }
    // Normal tipped-order fixtures fund all default pools; shortage is an explicit fixture.
    if (hasOrders && hasTips && options.fundingShortfall !== true) tipCents = Math.max(tipCents, Math.round(salesCents * 0.2));
    var surchargeCents = hasOrders ? hash(key + ':surcharge') % 1500 : 0;
    var orders = [];
    if (hasOrders) {
      // Vary attributable order counts without changing this employee's daily totals.
      var orderCount = 1 + hash(key + ':order-count') % 3;
      var remainingSales = salesCents, remainingTips = tipCents, remainingSurcharge = surchargeCents;
      for (var i = 0; i < orderCount; i++) {
        var remainingOrders = orderCount - i;
        var sale = Math.floor(remainingSales / remainingOrders);
        var tip = Math.floor(remainingTips / remainingOrders);
        var charge = Math.floor(remainingSurcharge / remainingOrders);
        remainingSales -= sale; remainingTips -= tip; remainingSurcharge -= charge;
        orders.push({ id: key + ':order:' + i, employeeId: employeeId, businessDate: options.dateKey,
          salesAmount: sale / 100, tipAmount: tip / 100, surcharge: charge / 100,
          tipType: i ? 'Cash Tips (现金小费)' : 'Credit Card Tips (信用卡小费)',
          orderType: i ? 'takeout' : 'dine_in', area: i ? 'bar' : 'main', paymentMethod: i ? 'cash' : 'card', localTime: i ? '19:30' : '11:30',
          completedPaymentAt: options.dateKey + 'T' + (i ? '19:30' : '11:30') + ':00' + chicagoOffset(options.dateKey), timeZone: 'America/Chicago',
          lines: [{ productId: i ? 'demo-beverage' : 'demo-main', productLine: 'POS', category: i ? '酒水' : '主菜', amount: sale / 100 }] });
      }
    }
    var punch = attendance(combination < 4, contributionScenario ? 0 : hash(key + ':attendance') % 4);
    punch.punchSessions = punch.punchSessions.map(function(item) { return datedSession(item, options.dateKey); });
    var reportedTipEvents = !hasOrders && tipCents ? [{ id: key + ':reported:0', employeeId: employeeId, businessDate: options.dateKey, amount: tipCents / 100,
      occurredAt: options.dateKey + 'T12:30:00' + chicagoOffset(options.dateKey), timeZone: 'America/Chicago' }] : [];
    return { version: VERSION, key: key, storeId: storeId, employeeId: employeeId, name: String(employee.name || ''), role: String(employee.role || ''),
      dateKey: options.dateKey, scenarioId: 'B0' + (combination + 1), contributionScenario: contributionScenario, attendance: punch,
      orders: orders, salesAmount: salesCents / 100, orderTips: hasOrders ? tipCents / 100 : 0,
      reportedTips: hasOrders ? 0 : tipCents / 100, reportedTipEvents: reportedTipEvents,
      originalTips: tipCents / 100, surcharge: surchargeCents / 100 };
  }
  function day(options) {
    required(options.storeId, 'storeId'); dateValue(options.dateKey);
    if (!Array.isArray(options.employees)) throw new Error('employees must be an array');
    var seen = Object.create(null);
    return options.employees.map(function(employee) {
      var id = required(employee && employee.id, 'employeeId');
      if (seen[id]) throw new Error('Duplicate employeeId: ' + id);
      seen[id] = true;
      return fact({ storeId: options.storeId, employee: employee, dateKey: options.dateKey, fundingShortfall: options.fundingShortfall });
    });
  }
  root.TipOutScenarioData = { version: VERSION, fact: fact, day: day, monthDates: monthDates };
})(typeof window !== 'undefined' ? window : this);
