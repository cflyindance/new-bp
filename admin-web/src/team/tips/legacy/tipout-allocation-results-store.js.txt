(function (root) {
  "use strict";

  var RESULT_KEY = "tipout_allocation_results_v1";
  var ALLOC_KEY = "tipout_allocated";

  function clone(value) { return value == null ? value : JSON.parse(JSON.stringify(value)); }
  function key(store, dateKey) { return String(store || "") + "\u0000" + String(dateKey || ""); }
  function parseDocument(storageKey) {
    var raw = localStorage.getItem(storageKey);
    if (!raw) return {};
    var parsed = JSON.parse(raw);
    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) throw new Error("本地分配数据已损坏");
    return parsed;
  }
  function allocatedDocument() { return parseDocument(ALLOC_KEY); }
  function resultDocument() { return parseDocument(RESULT_KEY); }
  function isAllocated(store, dateKey) {
    var dates = allocatedDocument()[String(store || "")];
    return Array.isArray(dates) && dates.indexOf(String(dateKey || "")) >= 0;
  }
  function read(store, dateKey) {
    if (!isAllocated(store, dateKey)) return null;
    var value = resultDocument()[key(store, dateKey)];
    return value && typeof value === "object" && !Array.isArray(value) ? clone(value) : null;
  }
  function finiteNonnegative(value) { return typeof value === "number" && isFinite(value) && value >= 0; }
  function validate(snapshot) {
    if (root.TipOutFunding) { try { root.TipOutFunding.validateSnapshot(snapshot); } catch (error) { return error.message; } }
    if (!snapshot || typeof snapshot !== "object") return "分配结果无效";
    if (!String(snapshot.store || "").trim()) return "请选择门店";
    if (!/^\d{4}-\d{2}-\d{2}$/.test(String(snapshot.dateKey || ""))) return "请选择有效日期";
    if (!Array.isArray(snapshot.pools) || !snapshot.pools.length) return "当前门店没有可用的小费分配规则";
    var invalid = false;
    function check(value) { if (!finiteNonnegative(value)) invalid = true; }
    (snapshot.pools || []).forEach(function (pool) {
      check(pool.poolAmount);
      (pool.employees || []).forEach(function (employee) {
        check(employee.hours); check(employee.percentage); check(employee.amount);
        if (employee.employeeWeight != null) check(employee.employeeWeight);
        if (!employee.employeeId || !employee.name || !employee.role) invalid = true;
      });
    });
    var summary = snapshot.summary || {};
    [summary.originalTips, summary.poolAmount, summary.allocatedAmount, summary.unallocatedAmount].forEach(check);
    return invalid ? "分配数据存在无效金额、工时、比例或员工" : null;
  }
  async function commit(snapshot, options) {
    var validationError = validate(snapshot);
    if (validationError) throw new Error(validationError);
    if (root.TipOutDateState) {
      return root.TipOutDateState.withLock(snapshot.store, snapshot.dateKey, async function () {
        root.TipOutDateState.assertDateWritable(snapshot.store, snapshot.dateKey);
        if (options && options.expectedSnapshotFingerprint !== undefined && JSON.stringify(read(snapshot.store, snapshot.dateKey)) !== options.expectedSnapshotFingerprint) throw new Error('分配结果已被其他页面修改，请刷新后重试');
        if (options && options.onlyIfUnallocated && isAllocated(snapshot.store, snapshot.dateKey)) throw new Error('当天已分配，请刷新查看结果');
        if (options && options.expectedSnapshotHash && (!read(snapshot.store, snapshot.dateKey) || read(snapshot.store, snapshot.dateKey).snapshotHash !== options.expectedSnapshotHash)) throw new Error('分配结果已被其他页面修改，请刷新后重试');
        var prepared = await root.TipOutDateState.prepareSnapshot(snapshot);
        return commitPrepared(prepared);
      });
    }
    if (options && options.onlyIfUnallocated) throw new Error('当前环境不支持安全快捷分配');
    return commitPrepared(snapshot);
  }
  function commitPrepared(snapshot) {
    var oldResultsRaw = localStorage.getItem(RESULT_KEY);
    var oldAllocatedRaw = localStorage.getItem(ALLOC_KEY);
    var results = resultDocument();
    var allocated = allocatedDocument();
    var store = String(snapshot.store);
    var dateKey = String(snapshot.dateKey);
    results[key(store, dateKey)] = clone(snapshot);
    var dates = Array.isArray(allocated[store]) ? allocated[store].slice() : [];
    if (dates.indexOf(dateKey) < 0) dates.push(dateKey);
    allocated[store] = dates;
    try {
      localStorage.setItem(RESULT_KEY, JSON.stringify(results));
      localStorage.setItem(ALLOC_KEY, JSON.stringify(allocated));
      if (root.TipOutDateState && root.TipOutDateState.clearUnconfirmedUpdate) root.TipOutDateState.clearUnconfirmedUpdate(store, dateKey);
    } catch (error) {
      try {
        if (oldResultsRaw == null) localStorage.removeItem(RESULT_KEY); else localStorage.setItem(RESULT_KEY, oldResultsRaw);
        if (oldAllocatedRaw == null) localStorage.removeItem(ALLOC_KEY); else localStorage.setItem(ALLOC_KEY, oldAllocatedRaw);
      } catch (_) {}
      throw error;
    }
    return clone(snapshot);
  }

  async function cancel(store, dateKey) {
    function execute() {
      if (root.TipOutDateState) root.TipOutDateState.assertDateWritable(store, dateKey);
      var oldResultsRaw = localStorage.getItem(RESULT_KEY), oldAllocatedRaw = localStorage.getItem(ALLOC_KEY);
      var results = resultDocument();
      var allocated = allocatedDocument();
      delete results[key(store, dateKey)];
      var dates = Array.isArray(allocated[String(store || '')]) ? allocated[String(store || '')].slice() : [];
      allocated[String(store || '')] = dates.filter(function (item) { return item !== String(dateKey || ''); });
      try {
        localStorage.setItem(RESULT_KEY, JSON.stringify(results));
        localStorage.setItem(ALLOC_KEY, JSON.stringify(allocated));
      } catch (error) {
        try {
          if (oldResultsRaw == null) localStorage.removeItem(RESULT_KEY); else localStorage.setItem(RESULT_KEY, oldResultsRaw);
          if (oldAllocatedRaw == null) localStorage.removeItem(ALLOC_KEY); else localStorage.setItem(ALLOC_KEY, oldAllocatedRaw);
        } catch (_) {}
        throw error;
      }
      if (root.TipOutDateState) root.TipOutDateState.notify(store, dateKey);
    }
    if (root.TipOutDateState) return root.TipOutDateState.withLock(store, dateKey, execute);
    execute();
  }

  var CANCEL_HISTORY_KEY = 'tipout-allocation-cancel-history-v1';
  function cancelFingerprint(store, dateKey) {
    if (!root.TipOutDateState) throw new Error('取消分配状态服务尚未就绪');
    return JSON.stringify([isAllocated(store,dateKey),root.TipOutDateState.readAllocation(store,dateKey),
      root.TipOutDateState.readPayout(store,dateKey),root.TipOutDateState.readUnconfirmedUpdate(store,dateKey)]);
  }
  async function resetConfirmed(store, dateKey, options) {
    if (!options || options.confirmed !== true || typeof options.expectedFingerprint !== 'string') throw new Error('请先确认取消分配');
    if (!String(store||'').trim() || !/^\d{4}-\d{2}-\d{2}$/.test(String(dateKey||''))) throw new Error('取消分配范围无效');
    return root.TipOutDateState.withLock(store,dateKey,function() {
      if (cancelFingerprint(store,dateKey)!==options.expectedFingerprint) throw new Error('分配或发放数据已变化，请重新确认');
      var state=root.TipOutDateState.inspect(store,dateKey);
      if (state.error) throw new Error(state.error);
      if (!state.snapshot && !state.payout && !isAllocated(store,dateKey)) return false;
      var historyRaw=localStorage.getItem(CANCEL_HISTORY_KEY);
      var history=historyRaw ? JSON.parse(historyRaw) : [];
      if (!Array.isArray(history)) throw new Error('取消分配历史已损坏');
      var results=resultDocument(),allocated=allocatedDocument();
      var payouts=parseDocument(root.TipOutDateState.PAYOUT_KEY),updates=parseDocument(root.TipOutDateState.UPDATE_KEY);
      history.push({recordId:root.TipOutDateState.requestId(),storeId:String(store),businessDate:String(dateKey),
        canceledAt:new Date().toISOString(),source:'summary-demo-reset',snapshot:state.snapshot,payout:state.payout,
        unconfirmedUpdate:root.TipOutDateState.readUnconfirmedUpdate(store,dateKey)});
      delete results[key(store,dateKey)];delete payouts[key(store,dateKey)];delete updates[key(store,dateKey)];
      allocated[String(store)]=(allocated[String(store)]||[]).filter(function(d){return d!==dateKey;});
      var writes=[[CANCEL_HISTORY_KEY,history],[RESULT_KEY,results],[ALLOC_KEY,allocated],
        [root.TipOutDateState.PAYOUT_KEY,payouts],[root.TipOutDateState.UPDATE_KEY,updates]];
      var before=writes.map(function(w){return localStorage.getItem(w[0]);});
      try { writes.forEach(function(w){localStorage.setItem(w[0],JSON.stringify(w[1]));}); }
      catch(error) {
        var rollbackError=null;
        writes.forEach(function(w,i){try{if(before[i]==null)localStorage.removeItem(w[0]);else localStorage.setItem(w[0],before[i]);}catch(e){rollbackError=e;}});
        if(rollbackError)throw new Error('取消失败且回滚不完整，请停止操作并检查本地数据');
        throw error;
      }
      root.TipOutDateState.notify(store,dateKey);
      return true;
    });
  }
  root.TipOutAllocationResults = { RESULT_KEY: RESULT_KEY, CANCEL_HISTORY_KEY:CANCEL_HISTORY_KEY,
    read: read, isAllocated: isAllocated, validate: validate, commit: commit, cancel: cancel,
    cancelFingerprint:cancelFingerprint,resetConfirmed:resetConfirmed };
})(window);
